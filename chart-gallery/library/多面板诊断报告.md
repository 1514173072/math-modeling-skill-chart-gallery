## 图表说明

一页同时回答“原始数据怎样、质量风险在哪、时间依赖多强、模型残差是否有结构、下一步做什么”。各面板必须共享同一分析样本和口径；结论框只复述代码已计算的指标，不能由模型凭视觉自由生成。仪表盘适合诊断汇报，不应代替完整统计检验和逐项附录。

```python
# -*- coding: utf-8 -*-
# 图表模板：多面板诊断报告（时序、质量、ACF、残差、计算型摘要）
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False, "font.size": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight",
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00"]

# ===== 1. 合成示例数据（替换为同一分析样本的真实输入） =====
rng = np.random.default_rng(21)
t = np.arange(240)
y1 = np.sin(2 * np.pi * t / 24) + 0.004 * t + rng.normal(0, 0.28, len(t))
y2 = 0.55 * np.roll(y1, 2) + rng.normal(0, 0.34, len(t))
y3 = np.cos(2 * np.pi * t / 48) + rng.normal(0, 0.25, len(t))
series = {"序列甲": y1, "序列乙": y2, "序列丙": y3}
fitted = 0.82 * y1
resid = y1 - fitted
quality_names = ["缺失", "异常", "断点", "漂移", "非高斯"]
quality = np.array([0.03, 0.11, 0.00, 0.36, 0.48])

def acf(x, max_lag=30):
    x = x - np.mean(x)
    denom = np.dot(x, x)
    return np.array([np.dot(x[:-k], x[k:]) / denom if k else 1.0
                     for k in range(max_lag + 1)])

# ===== 2. 组合版式 =====
fig, axd = plt.subplot_mosaic(
    [["ts", "ts", "quality"], ["acf", "resid", "summary"]],
    figsize=(12.5, 7.0), constrained_layout=True,
)
for (name, values), color in zip(series.items(), OI):
    axd["ts"].plot(t, values, lw=1.15, color=color, label=name)
axd["ts"].set_xlabel("时间")
axd["ts"].set_ylabel("标准化观测")
axd["ts"].legend(ncols=3, loc="upper left")

colors = [OI[3] if v >= 0.30 else "#7A8A99" for v in quality]
axd["quality"].barh(quality_names, quality, color=colors)
axd["quality"].axvline(0.30, color="#555555", ls="--", lw=0.9)
axd["quality"].set_xlim(0, 0.6)
axd["quality"].set_xlabel("风险分数")
axd["quality"].invert_yaxis()

lags = np.arange(31)
for (name, values), color in zip(series.items(), OI):
    axd["acf"].plot(lags, acf(values), lw=1.5, color=color, label=name)
axd["acf"].axhline(0, color="#777777", lw=0.8)
axd["acf"].set_xlabel("滞后阶数")
axd["acf"].set_ylabel("自相关")

axd["resid"].scatter(fitted, resid, s=13, alpha=0.55, color=OI[0], edgecolors="none")
axd["resid"].axhline(0, color="#777777", lw=0.9)
axd["resid"].set_xlabel("拟合值")
axd["resid"].set_ylabel("残差")

high_risk = [n for n, v in zip(quality_names, quality) if v >= 0.30]
lag1 = acf(y1, 1)[1]
summary = (
    f"计算摘要\n\n"
    f"高风险维度：{'、'.join(high_risk) if high_risk else '无'}\n"
    f"序列甲 lag-1 ACF：{lag1:.2f}\n"
    f"残差均值：{resid.mean():.3f}\n"
    f"残差标准差：{resid.std(ddof=1):.3f}\n\n"
    "建议：先处理高风险维度，\n再检查残差的时间结构。"
)
axd["summary"].text(
    0.05, 0.95, summary, transform=axd["summary"].transAxes,
    va="top", linespacing=1.5,
    bbox=dict(boxstyle="round,pad=0.6", facecolor="#F3F6F8", edgecolor="#9AA6B2"),
)
axd["summary"].set_axis_off()

fig.savefig("多面板诊断报告.png")
plt.close(fig)
print("已输出：多面板诊断报告.png")
```
