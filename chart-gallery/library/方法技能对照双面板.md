## 图表说明

两种方法在同一批对象上"谁更靠谱"的完整对照：左象限图暴露逐样本分歧（同号象限=两法同向），点色带出流量权重；右条形图给出与实测一致率的汇总，二项 CI 对照 50% 无信息线，加权与不加权并排互证。适合 CCM/RCM、两模型两评分器的对比评估。注意：象限底色只是视觉分区，文本标签放角落；加权一致率的 CI 需 bootstrap，图中可只给不加权 CI。

```python
# -*- coding: utf-8 -*-
# 图表模板：方法一致性对照双面板（象限散点 + 一致率条形）
# 数据契约：两方法得分差 + 逐样本一致标记(0/1) + 权重；需 scipy
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.patches as mpatches
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
from scipy import stats

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT, C_GREY = "#0072B2", "#D55E00", "#6B6B6B"

# ===== 1. 合成示例数据（使用时替换为真实输出） =====
rng = np.random.default_rng(26)
n = 40
ccm_gap = rng.normal(0.04, 0.05, n)                       # 方法甲技能差
rcm_gap = 0.8 * ccm_gap + rng.normal(0.03, 0.04, n)       # 方法乙技能差（正相关）
flow = rng.lognormal(9, 1.2, n)                           # 每对双向流量
ccm_match = (ccm_gap > 0).astype(int)
rcm_match = (rcm_gap > 0).astype(int)
df = pd.DataFrame({"ccm_gap": ccm_gap, "rcm_gap": rcm_gap,
                   "flow": flow, "ccm_match": ccm_match, "rcm_match": rcm_match})

# ===== 2. 绘图 =====
fig, axes = plt.subplots(1, 2, figsize=(11.5, 4.8),
                         gridspec_kw={"width_ratios": [1.15, 1]})

ax = axes[0]
sc = ax.scatter(df["ccm_gap"], df["rcm_gap"], s=46, c=np.log1p(df["flow"]),
                cmap="cividis", alpha=0.82, edgecolors="white", lw=0.6)
ax.axhline(0, color="#777", lw=0.9)
ax.axvline(0, color="#777", lw=0.9)
xr, yr = ax.get_xlim(), ax.get_ylim()
BLUE, ORANGE = "#0072B2", "#D55E00"
ax.add_patch(mpatches.Rectangle((0, 0), xr[1], yr[1], color=BLUE, alpha=0.05, zorder=0))
ax.add_patch(mpatches.Rectangle((xr[0], yr[0]), -xr[0], -yr[0], color=BLUE, alpha=0.05, zorder=0))
ax.add_patch(mpatches.Rectangle((0, yr[0]), xr[1], -yr[0], color=ORANGE, alpha=0.04, zorder=0))
ax.add_patch(mpatches.Rectangle((xr[0], 0), -xr[0], yr[1], color=ORANGE, alpha=0.04, zorder=0))
ax.text(0.97, 0.95, "两法方向一致（Ⅰ）", transform=ax.transAxes, ha="right", fontsize=9, color="#666")
ax.text(0.03, 0.04, "两法方向一致（Ⅲ）", transform=ax.transAxes, fontsize=9, color="#666")
ax.set_xlabel("方法甲 技能差 s(a→b) − s(b→a)")
ax.set_ylabel("方法乙 技能差 s(a→b) − s(b→a)")
cb = fig.colorbar(sc, ax=ax, shrink=0.8)
cb.set_label("ln(1+权重)", fontsize=8.5)

ax = axes[1]
bars = [("方法甲", df["ccm_match"].sum(), C_GREY),
        ("方法乙", df["rcm_match"].sum(), C_MAIN),
        ("方法甲（加权）", np.average(df["ccm_match"], weights=df["flow"]), "#B9C7D6"),
        ("方法乙（加权）", np.average(df["rcm_match"], weights=df["flow"]), "#7FA8CC")]
for i, (name, v, cc) in enumerate(bars):
    v = float(v) / n if v > 1.5 else float(v)
    p = stats.binomtest(int(round(v * n)), n, 0.5).pvalue if i < 2 else np.nan
    ci = stats.binomtest(int(round(v * n)), n).proportion_ci(0.95) if i < 2 else None
    err = (ci.high - ci.low) / 2 if ci else 0
    ax.bar(i, v * 100, 0.62, color=cc,
           yerr=err * 100 if err else None, error_kw=dict(ecolor="#555", lw=1.2, capsize=3))
    txt = f"{v:.0%}" + (f"\np={p:.3f}" if not np.isnan(p) else "")
    ax.text(i, v * 100 + 4, txt, ha="center", fontsize=9)
ax.axhline(50, color="#999", lw=1.0, ls="--")
ax.set_xticks(range(4), [b[0] for b in bars], fontsize=8.6)
ax.set_ylabel("与实测方向一致率（%）")
ax.set_ylim(0, 100)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("方法技能对照双面板.png")
plt.close(fig)
print(f"已输出：方法技能对照双面板.png（{n} 对）")
```
