## 图表说明

洛伦兹曲线把单元按强度升序累计，离 45° 均匀线越远越集中；基尼系数给出单一标尺，左右面板一个截面、一个时变，回答"集中度在加剧还是缓解"。适合站点/门店/用户的份额集中分析，也适用于收入、访问量等任何非负截面分布。注意：单元量级大时排序求和先转 float64 防 int32 溢出（模板已处理）。

```python
# -*- coding: utf-8 -*-
# 图表模板：洛伦兹曲线与基尼系数（含时变面板）
# 数据契约：截面=N个单元强度值(可多组对比)；时变=按期×单元面板算逐期基尼
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]
C_MAIN, C_ACCENT, C_GREY = OI[0], OI[3], "#6B6B6B"

# ===== 1. 工具函数（通用数学，可直接复用） =====
def gini(x: np.ndarray) -> float:
    x = np.sort(np.asarray(x, dtype=np.float64))   # 防 int 溢出
    n = len(x)
    return (2 * np.arange(1, n + 1) - n - 1).dot(x) / (n * x.sum())

def lorenz(x: np.ndarray):
    xs = np.sort(np.asarray(x, dtype=np.float64))
    cum = np.concatenate([[0.0], np.cumsum(xs) / xs.sum()])
    share = np.arange(len(cum)) / (len(cum) - 1)
    return share, cum

# ===== 2. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(7)
n_unit = 180
s2024 = rng.lognormal(8.2, 1.05, n_unit)     # sigma 越大越集中
s2025 = rng.lognormal(8.2, 1.18, n_unit)

# ===== 3. 绘图 =====
fig, axes = plt.subplots(1, 2, figsize=(11, 4.4))
ax = axes[0]
for s, cc, y in [(s2024, C_MAIN, "2024"), (s2025, C_ACCENT, "2025")]:
    share, cum = lorenz(s)
    ax.plot(share * 100, cum * 100, color=cc, lw=2.2, label=f"{y} 年（G={gini(s):.3f}）")
ax.plot([0, 100], [0, 100], "--", color=C_GREY, lw=1.2)
ax.text(38, 44, "均匀分布线", rotation=45, color=C_GREY, fontsize=9)
ax.set_xlabel("单元累计占比（%，按强度升序）")
ax.set_ylabel("强度累计占比（%）")
ax.legend(loc="upper left")
top20 = np.sort(s2025)[::-1][:n_unit // 5].sum() / s2025.sum()
ax.annotate(f"Top 20% 单元承载 {top20:.0%} 强度", xy=(80, top20 * 100), xytext=(30, 88),
            arrowprops=dict(arrowstyle="->", color="#444", lw=0.9), fontsize=9.5)

ax = axes[1]
sig = np.linspace(1.05, 1.18, 24) + rng.normal(0, 0.015, 24)
gm = np.array([gini(rng.lognormal(8.2, s, n_unit)) for s in sig])
ax.plot(range(24), gm, color=C_MAIN, lw=1.8)
ax.fill_between(range(24), gm, 0.5, color=C_MAIN, alpha=0.10)
ax.axhline(gm.mean(), color=C_ACCENT, ls="--", lw=1.2)
ax.text(0.5, gm.mean() + 0.004, f"两年均值 {gm.mean():.3f}", color=C_ACCENT, fontsize=9)
ax.set_xticks(range(0, 24, 4), [f"第 {m + 1} 月" for m in range(0, 24, 4)], fontsize=8.5)
ax.set_xlabel("月份")
ax.set_ylabel("单元基尼系数")
ax.set_ylim(0.5, None)

# ===== 4. 输出（文件名必须与笔记同名） =====
fig.savefig("洛伦兹曲线与基尼.png")
plt.close(fig)
print("已输出：洛伦兹曲线与基尼.png")
```
