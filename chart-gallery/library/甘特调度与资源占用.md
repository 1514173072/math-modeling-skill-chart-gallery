## 图表说明

上图按资源泳道展示任务时段，橙边表示关键任务；下图汇总同时运行的任务数并标出容量上限。图只能展示求解器输出的真实时间区间，不能为了版面整齐而移动任务；若有前置关系，应在正文或网络图中另行核验。

```python
# -*- coding: utf-8 -*-
# 数据契约：任务名、资源、开始、结束、是否关键
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#D9D9D9","savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73","#D55E00","#CC79A7","#56B4E9","#F0E442","#999999"]
tasks=[("A1","计算单元A",0,4,1),("A2","计算单元A",5,9,0),("A3","计算单元A",11,15,1),("B1","计算单元B",1,6,0),("B2","计算单元B",7,12,1),("B3","计算单元B",13,17,0),("C1","存储通道",0,3,0),("C2","存储通道",4,8,1),("C3","存储通道",9,14,0),("C4","存储通道",15,18,1)]
resources=["计算单元A","计算单元B","存储通道"]; ymap={r:i for i,r in enumerate(resources)}
fig,(ax,occ)=plt.subplots(2,1,figsize=(9.4,6.1),sharex=True,gridspec_kw={"height_ratios":[3,1],"hspace":.10})
for j,(name,res,start,end,critical) in enumerate(tasks):
    y=ymap[res]; ax.barh(y,end-start,left=start,height=.52,color=OI[y],alpha=.82,edgecolor=OI[1] if critical else "white",linewidth=2 if critical else .8)
    ax.text((start+end)/2,y,name,ha="center",va="center",color="white",fontsize=8,weight="bold")
ax.set_yticks(range(len(resources)),resources); ax.invert_yaxis(); ax.set_ylabel("资源泳道")
grid=np.linspace(0,18,181); active=np.array([sum(s<=t<e for _,_,s,e,_ in tasks) for t in grid])
occ.step(grid,active,where="post",color=OI[3],lw=2); occ.fill_between(grid,active,step="post",color=OI[3],alpha=.14)
occ.axhline(3,color="#555555",ls="--",lw=1,label="并发容量上限"); occ.set(xlabel="时间",ylabel="并发数",ylim=(0,3.8)); occ.legend(loc="upper right")
fig.savefig("甘特调度与资源占用.png"); plt.close(fig)
print("已输出：甘特调度与资源占用.png")
```
