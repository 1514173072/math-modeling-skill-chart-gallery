## 图表说明

左曲线回答"剂量多大最有利"，右热图回答"效应持续几天、滞后多久显形"——非持久剂量响应（广告、天气、价格调整）的完整画像。色标用发散 RdBu、中性 1 居中，读图即知方向。注意：曲线置信带若来自 bootstrap，勿用近似缩放带冒充；热图行列网格要注明插值与否，未建模的滞后格留白。

```python
# -*- coding: utf-8 -*-
# 图表模板：剂量响应曲线 + 滞后效应热图（双面板）
# 数据契约：左=剂量网格效应曲线（比值型）；右=滞后×剂量效应矩阵，中性值 1
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT = "#0072B2", "#D55E00"

# ===== 1. 合成示例数据（使用时替换为真实模型输出） =====
temp = np.linspace(0, 35, 36)
irr_total = 0.50 + 0.85 * np.exp(-((temp - 22) ** 2) / (2 * 7.5 ** 2))   # 峰值约 1.35 @22
lags = np.arange(4)
irr_lag = 1 + (irr_total[None, :] - 1) * np.exp(-lags[:, None] / 1.6)    # 指数衰减剖面

# ===== 2. 绘图 =====
fig = plt.figure(figsize=(11, 4.4))
gs = fig.add_gridspec(1, 2, width_ratios=[1.15, 1], wspace=0.24)

ax = fig.add_subplot(gs[0])
ax.axhline(1.0, color="#999", lw=1.0, ls="--")
ax.plot(temp, irr_total, color=C_MAIN, lw=2.4)
ax.fill_between(temp, np.exp(np.log(irr_total) * 0.92),
                np.exp(np.log(irr_total) * 1.08), color=C_MAIN, alpha=0.10)
ax.axvline(20, color=C_ACCENT, lw=0.9, ls=":")
ax.text(20.4, irr_total.max() * 0.97, "参考剂量 20", fontsize=9, color=C_ACCENT)
peak_t = temp[irr_total.argmax()]
ax.annotate(f"峰值剂量 {peak_t:.0f}", xy=(peak_t, irr_total.max()),
            xytext=(peak_t + 4, irr_total.max() * 1.02),
            arrowprops=dict(arrowstyle="->", color="#444", lw=0.9), fontsize=9.5)
ax.set_xlabel("剂量（自变量水平）")
ax.set_ylabel("效应比 IRR（相对参考剂量）")

ax = fig.add_subplot(gs[1])
im = ax.imshow(irr_lag, aspect="auto", cmap="RdBu_r", vmin=0.55, vmax=1.45,
               origin="lower", extent=[temp.min(), temp.max(), -0.5, 3.5])
ax.set_xlabel("剂量（自变量水平）")
ax.set_ylabel("滞后（天）")
ax.set_yticks([0, 1, 2, 3], ["当日", "+1 日", "+2 日", "+3 日"])
cb = fig.colorbar(im, ax=ax, shrink=0.9)
cb.set_label("滞后特定 IRR")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("剂量响应与滞后热图.png")
plt.close(fig)
print("已输出：剂量响应与滞后热图.png")
```
