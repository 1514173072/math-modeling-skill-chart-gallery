## 图表说明

消融与敏感性曲线的精致画法：实心点+误差棒是各档位的多抖动均值，连线给出整体趋势，最关键的一步是把跌破零基准的区段向浅红着色——"贡献从哪一点起变成负担"不用读数就能定位。与《超参稳健性曲线》分工不同：那张看"选哪个超参"，这张看"某个组件/输入在多大强度内还有正贡献"。零基准线务必画成虚线并显眼；端点数值直标能让读者跳过坐标轴反查。负区着色只在 y<0 区间填充，别整幅打底。

```python
# -*- coding: utf-8 -*-
# 图表模板：收益衰减曲线（误差棒 + 零基准 + 负区着色）
# 数据契约：有序控制变量网格 × 均值收益 × 误差（抖动标准差），差值型指标
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG

import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]
C_MAIN, C_NEG = OI[0], OI[3]

# ===== 1. 合成示例数据（使用时替换为真实消融/敏感性结果） =====
rng = np.random.default_rng(15)
x = np.linspace(0, 100, 201)
y = 44 * np.exp(-x / 25) - 6 + 1.2 * np.sin(x / 8)          # 平滑均值曲线
xs = np.arange(0, 101, 10)                                   # 误差棒档位
ym = 44 * np.exp(-xs / 25) - 6 + 1.2 * np.sin(xs / 8) + rng.normal(0, 0.6, len(xs))
yerr = rng.uniform(1.4, 3.2, len(xs))
x_cross = x[np.argmax(y < 0)]                                # 首个跌破零的点

# ===== 2. 绘图 =====
fig, ax = plt.subplots(figsize=(9.2, 4.9))
ax.fill_between(x, y, 0, where=y < 0, color=C_NEG, alpha=0.18, label="负收益区")
ax.plot(x, y, color=C_MAIN, lw=1.8, label="均值收益")
ax.errorbar(xs, ym, yerr=yerr, fmt="o", color=C_MAIN, ms=4.5,
            capsize=3, lw=1.2, label="多种子均值±标准差")
ax.axhline(0, color="#6B6B6B", ls="--", lw=1.1)
ax.axvline(x_cross, color=C_NEG, ls=":", lw=1.1)
ax.text(x_cross + 1.5, 30, f"零收益点 ≈ {x_cross:.0f}%", color=C_NEG, fontsize=9.5)
ax.annotate(f"+{ym[0]:.0f}%", (xs[0], ym[0]), textcoords="offset points",
            xytext=(6, 7), color=C_MAIN, fontsize=9.5)
ax.annotate(f"{ym[-1]:.0f}%", (xs[-1], ym[-1]), textcoords="offset points",
            xytext=(-9, -3), ha="right", color=C_MAIN, fontsize=9.5)
ax.set_xlim(-2, 104)
ax.set_xlabel("上下文掩码率（%）")
ax.set_ylabel("相对固定文本的收益（%）")
ax.legend(loc="upper right")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("衰减曲线负区着色.png")
plt.close(fig)
print("已输出：衰减曲线负区着色.png")
```
