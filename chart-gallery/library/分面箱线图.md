## 图表说明

每个面板叠加箱线、抖动原始点和同一个体的前后连线。箱体概括分布，点和连线保留样本量与变化方向，能识别总体上升是否由少数个体驱动。所有面板共享纵轴。若数据并非配对，应删除连线；样本很少时以点图为主。

```python
# -*- coding: utf-8 -*-
# 图表模板：分面对比箱线图（多单元×前后测）
# 数据契约：长表(单元, 条件, 数值)；每面两箱、共享纵轴
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(2)
n_unit, n_obs = 8, 30
data = []
for i in range(n_unit):
    pre = rng.normal(2.2 + 0.06 * i, 0.5, n_obs)
    post = pre + rng.normal(0.7, 0.45, n_obs)        # 条件效应 + 单元噪声
    data.append((pre, post))

# ===== 2. 绘图 =====
fig, axes = plt.subplots(2, 4, figsize=(12.5, 5.8), sharey=True)
for ax, i in zip(axes.flat, range(n_unit)):
    pre, post = data[i]
    bp = ax.boxplot([pre, post], labels=["前测", "后测"], widths=0.55,
                    patch_artist=True, showfliers=False, medianprops=dict(color="#222"))
    for patch, cc in zip(bp["boxes"], [OI[7], OI[0]]):
        patch.set_facecolor(cc)
        patch.set_alpha(0.42)
    jitter = rng.normal(0, 0.035, n_obs)
    for j in range(n_obs):
        ax.plot([1 + jitter[j], 2 + jitter[j]], [pre[j], post[j]],
                color="#A7A7A7", lw=0.45, alpha=0.35, zorder=1)
    ax.scatter(1 + jitter, pre, s=9, color=OI[7], alpha=0.65, zorder=2)
    ax.scatter(2 + jitter, post, s=9, color=OI[0], alpha=0.65, zorder=2)
    ax.set_title(f"单元 {i + 1}（n={n_obs} 对）", fontsize=9.5)
    ax.grid(axis="y", alpha=0.5)
axes[0, 0].set_ylabel("强度")
for ax in axes[1]:
    ax.set_xlabel("条件")
fig.align_ylabels(axes[:, 0])

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("分面箱线图.png")
plt.close(fig)
print("已输出：分面箱线图.png")
```
