## 图表说明

分位数预测的标准展示：区间带像扇形一样包裹实测线，越宽说明不确定性越大；下方残差条带把"哪里失手"钉出来（大残差条染强调色）。适合论文的预测章节与模型汇报。注意：经验覆盖率(实测落在 80% 带内的比例)应与名义水平对照报告；残差阈值取正常波动的 2–3 倍，避免满图红。

```python
# -*- coding: utf-8 -*-
# 图表模板：预测扇形图（分位数区间带 + 残差条带）
# 数据契约：时间索引 + 实测 + q10/q25/q50/q75/q90 五条分位数列；残差=实测−中位数
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT, C_GREY = "#0072B2", "#D55E00", "#6B6B6B"

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(11)
ts = pd.date_range("2025-11-24", periods=14 * 24, freq="h")
hour, dow = ts.hour.to_numpy(), ts.dayofweek.to_numpy()
shape = 900 * np.exp(-((hour - 8) ** 2) / 7.0) + 1500 * np.exp(-((hour - 17.5) ** 2) / 11.0) + 500
y = shape * np.where(dow < 5, 1.0, 0.72) * rng.lognormal(0, 0.06, len(ts))
holiday = np.where(ts.strftime("%Y-%m-%d") == "2025-11-27")[0]   # 节假日压低
y[holiday] *= 0.62
sigma = 0.09 * y + 120
q50 = y * (1 + rng.normal(0, 0.02, len(ts)))
q10, q25 = q50 - 1.2816 * sigma, q50 - 0.6745 * sigma
q75, q90 = q50 + 0.6745 * sigma, q50 + 1.2816 * sigma

# ===== 2. 绘图 =====
x = np.arange(len(ts))
fig, (ax, ax2) = plt.subplots(2, 1, figsize=(11.5, 6.2), height_ratios=[3, 1],
                              sharex=True, gridspec_kw={"hspace": 0.08})
ax.fill_between(x, q10, q90, color=C_MAIN, alpha=0.16, label="80% 区间")
ax.fill_between(x, q25, q75, color=C_MAIN, alpha=0.28, label="50% 区间")
ax.plot(x, q50, color=C_MAIN, lw=1.7, label="中位数预测")
ax.plot(x, y, color="#222", lw=1.1, label="实测")
ax.set_ylabel("系统强度（次/时）")
ax.legend(loc="upper left", ncols=4)
ax.annotate("节假日", xy=(int(holiday[0]) + 8, y[holiday].min() * 1.02),
            xytext=(int(holiday[0]) + 34, y.max() * 0.9), fontsize=9, color=C_ACCENT,
            arrowprops=dict(arrowstyle="->", color=C_ACCENT, lw=0.9))

resid = y - q50
ax2.bar(x, resid, color=[C_ACCENT if abs(r) > 600 else C_GREY for r in resid], width=0.85)
ax2.axhline(0, color="#555", lw=0.8)
ax2.set_ylabel("残差")
ticks = x[::24]
ax2.set_xticks(ticks, [t.strftime("%m-%d") for t in ts[::24]], fontsize=8.5)
ax2.set_xlabel("日期（2025 年）")

# ===== 3. 输出（文件名必须与笔记同名） =====
cov = ((y >= q10) & (y <= q90)).mean()
fig.savefig("预测扇形图.png")
plt.close(fig)
print(f"已输出：预测扇形图.png；80% 区间经验覆盖率 = {cov:.3f}")
```
