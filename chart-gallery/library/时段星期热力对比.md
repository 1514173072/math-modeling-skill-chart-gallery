## 图表说明

把"小时×星期"的强度矩阵按群体并排铺开、共用色标，结构差异（谁在通勤峰、谁在周末午间）不需要任何统计检验就能被读者看到。适合任何"两类主体×一天 24 小时"的行为对比。注意：两图必须共享 vmax，否则深浅不可比；面板名放在图内左上角白底小签，不算标题。

```python
# -*- coding: utf-8 -*-
# 图表模板：小时×星期热力对比（双群体共享色标）
# 数据契约：两组 7×24 矩阵（行=星期、列=小时、值=均值强度），共用 vmax
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(21)
hours = np.arange(24)
dows = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"]

def commuter(i):   # 工作日双峰、周末弱
    return (300 * np.exp(-((hours - 8) ** 2) / 4.5)
            + 360 * np.exp(-((hours - 17.5) ** 2) / 6) + 60) if i < 5 \
        else 130 * np.exp(-((hours - 13) ** 2) / 16) + 40

def leisure(i):    # 午间单峰、周末更强
    peak = 220 if i >= 5 else 110
    return peak * np.exp(-((hours - 13.5) ** 2) / 22) + 25

mat_m = pd.DataFrame(np.array([commuter(i) for i in range(7)])
                     * rng.normal(1, 0.05, (7, 24)), index=dows, columns=hours)
mat_c = pd.DataFrame(np.array([leisure(i) for i in range(7)])
                     * rng.normal(1, 0.05, (7, 24)), index=dows, columns=hours)

# ===== 2. 绘图 =====
fig, axes = plt.subplots(1, 3, figsize=(12.5, 4.4),
                         gridspec_kw={"width_ratios": [1, 1, 0.035], "wspace": 0.05})
vmax = max(mat_m.values.max(), mat_c.values.max())
for ax, mat, ttl, cm in [(axes[0], mat_m, "群体甲（通勤型）", "viridis"),
                         (axes[1], mat_c, "群体乙（休闲型）", "magma")]:
    im = ax.imshow(mat.values, aspect="auto", cmap=cm, vmin=0, vmax=vmax)
    ax.set_xticks(range(0, 24, 3), [f"{h}" for h in range(0, 24, 3)])
    ax.set_yticks(range(7), dows)
    ax.set_xlabel("小时")
    ax.invert_yaxis()
    ax.grid(False)
    ax.text(0.02, 0.97, ttl, transform=ax.transAxes, va="top", fontsize=11,
            bbox=dict(fc="white", ec="none", alpha=0.75, pad=2))
axes[0].set_ylabel("星期")
cb = fig.colorbar(im, cax=axes[2])
cb.set_label("日均强度（次/小时）")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("时段星期热力对比.png")
plt.close(fig)
print("已输出：时段星期热力对比.png")
```
