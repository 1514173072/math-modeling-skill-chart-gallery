## 图表说明

可解释性可视化的标准做法：普通折线只有一个颜色，把逐点归因（梯度/遮挡/注意力权重）编码成线段颜色后，模型"主要看哪一段"直接浮现在曲线上；两种归因方法并排，一致区段可信度高。适合深度时序模型（RC/LSTM/Transformer）的解释章节。注意：归因值要先对称归一化（正负同幅）；分段着色用 LineCollection 而非逐点 scatter（线才连续）；蓝橙对撞色板已按色盲友好调过。

```python
# -*- coding: utf-8 -*-
# 图表模板：归因着色时序图（可解释性双面板）
# 数据契约：一条时序 + 每方法逐点归因值（正负对称）
# 协议：全中文标注；图内不加"XX图"标题；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from matplotlib.collections import LineCollection
from matplotlib.colors import LinearSegmentedColormap, TwoSlopeNorm

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
CMAP = LinearSegmentedColormap.from_list("attr", ["#D55E00", "#E3E3E3", "#0072B2"])

# ===== 1. 合成示例数据（使用时替换为真实归因输出） =====
rng = np.random.default_rng(14)
t = np.arange(160)
series = 0.5 * np.sin(t / 9) + 0.25 * np.sin(t / 23) + rng.normal(0, 0.08, 160)
bump = np.exp(-((t - 70) ** 2) / 120) + 0.7 * np.exp(-((t - 110) ** 2) / 200)
attr_a = bump + rng.normal(0, 0.05, 160)               # 方法甲：聚焦两处事件
attr_b = bump * rng.uniform(0.4, 1.0, 160) + rng.normal(0, 0.08, 160)  # 方法乙：更分散
attr_b[40:60] -= 0.25                                   # 负贡献段
vmax = max(np.abs(attr_a).max(), np.abs(attr_b).max())
norm = TwoSlopeNorm(vmin=-vmax, vcenter=0, vmax=vmax)


def colored_line(ax, y, attr, label):
    pts = np.array([t, y]).T.reshape(-1, 1, 2)
    segs = np.concatenate([pts[:-1], pts[1:]], axis=1)
    ax.plot(t, y, color="#B9C2CB", lw=0.7, alpha=0.6, zorder=1)
    lc = LineCollection(segs, cmap=CMAP, norm=norm, linewidths=2.2)
    lc.set_array((attr[:-1] + attr[1:]) / 2)
    ax.add_collection(lc)
    ax.set_xlim(0, 160)
    ax.set_ylim(y.min() - 0.4, y.max() + 0.4)
    ax.set_title(label, fontsize=10.5)
    ax.set_xlabel("时间步")
    ax.set_ylabel("归一化强度")


# ===== 2. 绘图 =====
fig, axes = plt.subplots(2, 1, figsize=(11.5, 6.6), sharex=True, sharey=True)
colored_line(axes[0], series, attr_a, "方法甲（梯度型）")
colored_line(axes[1], series, attr_b, "方法乙（遮挡型）")
cb = fig.colorbar(plt.cm.ScalarMappable(cmap=CMAP, norm=norm), ax=axes, shrink=0.9)
cb.set_label("归因贡献（蓝=正 / 橙=负）")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("归因着色时序图.png")
plt.close(fig)
print("已输出：归因着色时序图.png")
```
