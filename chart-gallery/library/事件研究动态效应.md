## 图表说明

事件研究的标准呈现：事件前系数应贴着零（平行趋势可视化），事件日跳变、随后衰减的路径即动态效应；三类事件横向微错开（±0.18 天）避免点重叠；灰色安慰剂带来自随机化推断，效应只有探出灰带才算真。适合政策/极端事件/产品动作的效应评估。注意：事件间错开量要远小于 1 天；纵轴写明"log 尺度 95% CI"。

```python
# -*- coding: utf-8 -*-
# 图表模板：事件研究动态效应图（多事件 + 安慰剂带）
# 数据契约：各事件 rel/coef/lo/hi/n_events + 安慰剂 null 分位数；纵轴 (exp(β)−1)×100
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_GREY = "#6B6B6B"

# ===== 1. 合成示例数据（使用时替换为真实回归输出） =====
rng = np.random.default_rng(23)
rel = np.arange(-7, 8)
rows = []
for ev, impact, decay, se, n_ev, cc in [
        ("事件甲", -0.38, 1.6, 0.045, 22, "#2166AC"),
        ("事件乙", -0.14, 2.2, 0.040, 15, "#B2182B"),
        ("事件丙", -0.10, 1.2, 0.038, 38, "#4393C3")]:
    coef = np.where(rel >= 0, impact * np.exp(-np.maximum(rel, 0) / decay), 0.0)
    coef += rng.normal(0, 0.012, len(rel)) * (rel < 0)     # 事前小抖动
    for r, c in zip(rel, coef):
        rows.append((ev, r, c, c - 1.96 * se, c + 1.96 * se, n_ev, cc))
es = pd.DataFrame(rows, columns=["事件", "rel", "coef", "lo", "hi", "n_events", "cc"])
null_lo, null_hi = -0.028, 0.030                          # 安慰剂 null 95% 分位数（log 尺度）

# ===== 2. 绘图 =====
fig, ax = plt.subplots(figsize=(10.5, 5.2))
offsets = {"事件甲": -0.18, "事件乙": 0.0, "事件丙": 0.18}
for ev, g in es.groupby("事件"):
    x = g["rel"] + offsets[ev]
    irr = (np.exp(g["coef"]) - 1) * 100
    lo, hi = (np.exp(g["lo"]) - 1) * 100, (np.exp(g["hi"]) - 1) * 100
    cc = g["cc"].iloc[0]
    ax.plot(x, irr, "-o", ms=3.4, lw=1.7, color=cc,
            label=f"{ev}（n={int(g['n_events'].iloc[0])}）")
    ax.fill_between(x, lo, hi, color=cc, alpha=0.13)

ax.axhline(0, color="#777", lw=0.9)
ax.axvline(0, color="#999", lw=0.9, ls=":")
ax.axhspan((np.exp(null_lo) - 1) * 100, (np.exp(null_hi) - 1) * 100,
           color=C_GREY, alpha=0.22, zorder=0)
ax.text(-6.8, (np.exp(null_lo) - 1) * 100 + 0.5, "安慰剂 null 95% 区间（随机化推断）",
        fontsize=8.5, color="#666")
ax.text(0.3, -34, "事件日", fontsize=9, color="#777")
ax.set_xlabel("相对事件日的天数")
ax.set_ylabel("需求变化（%，log 尺度 95% CI）")
ax.set_xticks(range(-7, 8))
ax.legend(loc="lower left")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("事件研究动态效应.png")
plt.close(fig)
print("已输出：事件研究动态效应.png")
```
