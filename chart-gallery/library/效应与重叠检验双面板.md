## 图表说明

正式效应估计的"证据+资格"双交代：左边给效应量和置信区间，右边用倾向得分分布回答"这个效应在共同支撑域内站不站得住"——两组分布不重叠的区域就是外推区，边界线画出来，加权后的 SMD/ESS 诊断贴在图角。适合匹配/加权类识别策略的主结果图。注意：重叠边界是预设规则不是数据看出来的；两直方图都用 density 保证可比。

```python
# -*- coding: utf-8 -*-
# 图表模板：处理效应 + 倾向得分重叠检验（双面板）
# 数据契约：效应估计表(结局,估计,CI) + 逐样本倾向概率与分组；重叠边界可选
# 协议：全中文标注；图内不加"XX图"标题；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})

# ===== 1. 合成示例数据（使用时替换为真实估计与 OOF 倾向得分） =====
rng = np.random.default_rng(8)
est = pd.DataFrame({
    "outcome": ["结局甲（持续期）", "结局乙（早期）"],
    "estimate": [-0.128, -0.086],
    "ci_low": [-0.181, -0.142],
    "ci_high": [-0.075, -0.030],
})
ps_treated = rng.beta(4.5, 2.0, 800)      # 处理组倾向得分偏高
ps_control = rng.beta(2.0, 4.5, 1200)     # 对照组偏低
overlap = [0.05, 0.95]
diag = {"最大绝对 SMD": 0.031, "对照ESS": 486.2}

# ===== 2. 绘图 =====
fig = plt.figure(figsize=(11.8, 6.2))
gs = fig.add_gridspec(1, 2, width_ratios=[1, 1.25], wspace=0.34)

ax0 = fig.add_subplot(gs[0, 0])
y = np.arange(len(est))
low = est["estimate"] - est["ci_low"]
high = est["ci_high"] - est["estimate"]
ax0.errorbar(est["estimate"], y, xerr=np.vstack([low, high]), fmt="o", color="#1d4ed8",
             ecolor="#64748b", capsize=4, linewidth=1.5)
ax0.axvline(0, color="#991b1b", linestyle="--", linewidth=1)
ax0.set_yticks(y, est["outcome"])
ax0.set_xlabel("交叉拟合 DR-ATT（95% 置信区间）")
ax0.grid(axis="x", alpha=0.2)
for yi, v, h in zip(y, est["estimate"], est["ci_high"]):
    ax0.text(h + 0.012, yi, f"{v:.3f}", va="center", fontsize=8.5)

ax1 = fig.add_subplot(gs[0, 1])
bins = np.linspace(0, 1, 21)
ax1.hist(ps_control, bins=bins, density=True, alpha=0.48, color="#0f766e", label="同期对照")
ax1.hist(ps_treated, bins=bins, density=True, alpha=0.48, color="#f59e0b", label="处理组")
for b in overlap:
    ax1.axvline(b, color="#991b1b", linestyle="--", linewidth=1)
ax1.set_xlabel("OOF 倾向概率")
ax1.set_ylabel("密度")
ax1.set_xlim(0, 1)
ax1.grid(axis="y", alpha=0.2)
ax1.legend(frameon=False)
ax1.text(0.02, 0.98, f"最大绝对 SMD：{diag['最大绝对 SMD']:.3f}\n对照 ESS：{diag['对照ESS']:.1f}",
         transform=ax1.transAxes, va="top", fontsize=8.5)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("效应与重叠检验双面板.png")
plt.close(fig)
print("已输出：效应与重叠检验双面板.png")
```
