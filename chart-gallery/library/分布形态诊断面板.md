## 图表说明

左侧雨云图把核密度、箱线统计和原始样本放在同一坐标中；右侧密度脊线强调组间分布形态。核密度只用于描述分布，带宽会影响峰形；样本很少时应优先展示原始点，不应把平滑曲线解释为真实概率结构。

```python
# -*- coding: utf-8 -*-
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#D9D9D9","grid.linewidth":.6,"axes.axisbelow":True,"savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73","#D55E00"]
rng=np.random.default_rng(26)
data={
    "方案甲":rng.normal(68,7,110),
    "方案乙":np.r_[rng.normal(61,5,72),rng.normal(78,3,38)],
    "方案丙":55+rng.gamma(3.0,5.0,110),
    "方案丁":rng.normal(73,4.5,110),
}

def smooth_density(x, grid):
    bw=max(np.std(x)*.28,1e-6)
    z=(grid[:,None]-x[None,:])/bw
    return np.exp(-.5*z*z).mean(1)/(bw*np.sqrt(2*np.pi))

fig,(ax,ar)=plt.subplots(1,2,figsize=(11.2,5.9),gridspec_kw={"width_ratios":[1.08,1]})
parts=ax.violinplot(list(data.values()),positions=np.arange(1,5),showextrema=False,widths=.72)
for body,c in zip(parts["bodies"],OI): body.set_facecolor(c); body.set_edgecolor(c); body.set_alpha(.25)
bp=ax.boxplot(list(data.values()),positions=np.arange(1,5),widths=.18,patch_artist=True,showfliers=False,medianprops={"color":"#222222","lw":1.4},whiskerprops={"color":"#555555"},capprops={"color":"#555555"})
for box,c in zip(bp["boxes"],OI): box.set_facecolor("white"); box.set_edgecolor(c); box.set_linewidth(1.3)
for i,((name,x),c) in enumerate(zip(data.items(),OI),1):
    jitter=rng.normal(0,.045,len(x)); ax.scatter(i+.28+jitter,x,s=9,color=c,alpha=.38,edgecolor="none")
ax.set_xticks(range(1,5),data.keys()); ax.set_ylabel("综合得分"); ax.set_title("雨云图：分布＋箱线统计＋原始样本",loc="left",weight="bold")
ax.text(.01,.02,"点保留样本信息；琴身宽度表示估计密度",transform=ax.transAxes,color="#666666",fontsize=9)

all_x=np.concatenate(list(data.values())); grid=np.linspace(all_x.min()-4,all_x.max()+4,260)
for j,((name,x),c) in enumerate(zip(data.items(),OI)):
    den=smooth_density(x,grid); den=den/den.max()*.72
    ar.fill_between(grid,j,j+den,color=c,alpha=.34); ar.plot(grid,j+den,color=c,lw=1.6)
    ar.plot([np.median(x),np.median(x)],[j,j+.72],color=c,lw=1,ls="--")
ar.set_yticks(np.arange(4)+.18,data.keys()); ar.set_xlabel("综合得分"); ar.set_title("密度脊线：偏态、多峰与长尾",loc="left",weight="bold"); ar.grid(axis="x"); ar.grid(axis="y",visible=False); ar.invert_yaxis()
fig.subplots_adjust(wspace=.25)
fig.savefig("分布形态诊断面板.png"); plt.close(fig)
print("已输出：分布形态诊断面板.png")
```
