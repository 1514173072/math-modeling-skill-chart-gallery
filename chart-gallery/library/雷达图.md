## 图表说明

雷达图只负责快速识别“哪里突出、哪里短板”，右侧点线图承担精确比较。所有指标必须统一转换到 0–100，且方向一致为“越高越优”；成本、风险等逆向指标要先反向。坐标范围固定从 0 开始，维度顺序按理论框架固定，不能为了形成漂亮轮廓随意调整。对象最多三个；需要比较显著性、置信区间或很多对象时，应改用点区间图或热力图。

```python
# -*- coding: utf-8 -*-
# 图表模板：标准化画像雷达图 + 精确比较点线图
# 数据契约：2–3 个对象 × 5–8 个 0–100 同向标准化指标
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "figure.dpi": 110, "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实标准化指标） =====
dims = ["创新能力", "稳健性", "可解释性", "计算效率", "泛化能力", "落地价值"]
groups = {
    "方案甲": np.array([82, 68, 75, 61, 78, 72]),
    "方案乙": np.array([69, 81, 63, 84, 71, 77]),
}

n = len(dims)
angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
closed_angles = np.r_[angles, angles[0]]

# ===== 2. 绘图：左侧看画像，右侧读精确差异 =====
fig = plt.figure(figsize=(11.2, 5.5))
gs = fig.add_gridspec(1, 2, width_ratios=[1.15, 1.0], wspace=0.36)
ax = fig.add_subplot(gs[0, 0], projection="polar")
ax.set_theta_offset(np.pi / 2)
ax.set_theta_direction(-1)
for (label, values), color in zip(groups.items(), OI):
    closed_values = np.r_[values, values[0]]
    ax.plot(closed_angles, closed_values, color=color, linewidth=2,
            marker="o", markersize=3.5, label=label)
    ax.fill(closed_angles, closed_values, color=color, alpha=0.08)
ax.set_xticks(angles, dims, fontsize=10)
ax.set_ylim(0, 100)
ax.set_yticks([20, 40, 60, 80, 100], ["20", "40", "60", "80", "100"], fontsize=8)
ax.set_rlabel_position(12)
ax.legend(loc="upper left", bbox_to_anchor=(-0.10, 1.12), ncols=2)

ax2 = fig.add_subplot(gs[0, 1])
y = np.arange(n)
first, second = groups.values()
for yi, a, b in zip(y, first, second):
    ax2.plot([a, b], [yi, yi], color="#B5BDC5", lw=1.5, zorder=1)
ax2.scatter(first, y, color=OI[0], s=42, label=list(groups)[0], zorder=2)
ax2.scatter(second, y, color=OI[1], s=42, label=list(groups)[1], zorder=2)
for yi, a, b in zip(y, first, second):
    ax2.text(a, yi - 0.16, f"{a}", ha="center", va="bottom", fontsize=8.2, color=OI[0])
    ax2.text(b, yi + 0.16, f"{b}", ha="center", va="top", fontsize=8.2, color=OI[1])
ax2.set_yticks(y, dims)
ax2.invert_yaxis()
ax2.set_xlim(0, 100)
ax2.set_xlabel("标准化得分（0–100，越高越优）")
ax2.grid(axis="x", color="#D9D9D9", linewidth=0.6)
ax2.spines[["top", "right", "left"]].set_visible(False)
ax2.tick_params(axis="y", length=0)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("雷达图.png")
plt.close(fig)
print("已输出：雷达图.png")
```
