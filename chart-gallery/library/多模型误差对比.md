## 图表说明

每个指标一个面板，点表示跨折次或重复实验的均值，横线表示 95% 置信区间。最优均值用强调色，其余模型标注相对差距。三个指标排序不一致本身就是结果。置信区间反映的是重复评测的不确定性；若只有一次测试集结果，应只画点而不能虚构区间。

```python
# -*- coding: utf-8 -*-
# 图表模板：多模型误差对比（三指标面板条形）
# 数据契约：模型×指标表，每行一个模型，列为 MAE/RMSE/MAPE 等同向误差指标
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN = "#0072B2"

# ===== 1. 合成示例数据（使用时替换为逐折评测结果） =====
rng = np.random.default_rng(19)
models = ["季节朴素", "日朴素", "SARIMA", "LSTM", "LightGBM", "LightGBM\n无天气"]
centers = {
    "mae": [812, 935, 704, 658, 512, 607],
    "rmse": [1180, 1340, 1010, 950, 780, 890],
    "mape": [0.29, 0.33, 0.26, 0.24, 0.19, 0.22],
}
rows = []
for model_i, model in enumerate(models):
    for fold in range(8):
        for metric, vals in centers.items():
            noise = 0.045 if metric != "mape" else 0.035
            rows.append((model, fold, metric, vals[model_i] * rng.normal(1, noise)))
raw = pd.DataFrame(rows, columns=["model", "fold", "metric", "value"])
summary = raw.groupby(["model", "metric"])["value"].agg(["mean", "std", "count"]).reset_index()
summary["half"] = 1.96 * summary["std"] / np.sqrt(summary["count"])
order = (summary[summary["metric"].eq("mae")].sort_values("mean")["model"].tolist())
metrics = [("mae", "MAE（次/时）"), ("rmse", "RMSE（次/时）"), ("mape", "MAPE")]

# ===== 2. 绘图 =====
fig, axes = plt.subplots(1, 3, figsize=(12.8, 5.2), sharey=True)
for ax, (col, ttl) in zip(axes, metrics):
    sub = summary[summary["metric"].eq(col)].set_index("model").loc[order]
    vals = sub["mean"].to_numpy()
    half = sub["half"].to_numpy()
    best_i = int(np.argmin(vals))
    colors = [C_MAIN if i == best_i else "#7A8A99" for i in range(len(vals))]
    y = np.arange(len(order))
    for i, (v, e, c) in enumerate(zip(vals, half, colors)):
        ax.errorbar(v, i, xerr=e, fmt="o", ms=5.5, capsize=3,
                    color=c, ecolor=c, lw=1.4)
        label = f"{v:.3f}" if col == "mape" else f"{v:.0f}"
        gap = "" if i == best_i else f"  +{(v / vals[best_i] - 1) * 100:.0f}%"
        ax.text(v + e, i, f"  {label}{gap}", va="center", fontsize=8.1)
    ax.set_yticks(y, order)
    ax.invert_yaxis()
    ax.set_title(ttl, fontsize=10.5)
    ax.set_xlabel("均值与 95% 置信区间")
    ax.margins(x=0.28)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("多模型误差对比.png")
plt.close(fig)
print("已输出：多模型误差对比.png")
```
