## 图表说明

VaR 是给定置信水平的损失分位点，CVaR 是超过 VaR 后的平均损失。右图直接显示机会约束的违反概率是否低于允许水平。两部分必须来自同一情景生成机制；情景过少时，尾部估计高度不稳定，应同时做重采样区间或敏感性分析。

```python
# -*- coding: utf-8 -*-
# 数据契约：方案×情景损失及约束量；预先声明置信水平与允许违反概率
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#D9D9D9","savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73","#D55E00","#CC79A7","#56B4E9","#F0E442","#999999"]
rng=np.random.default_rng(44); alpha=.95
loss=rng.gamma(3.0,7.2,3500)+rng.binomial(1,.06,3500)*rng.gamma(2,16,3500)
var=np.quantile(loss,alpha); cvar=loss[loss>=var].mean()
schemes={"稳健方案":rng.normal(72,9,2200),"均衡方案":rng.normal(78,12,2200),"激进方案":rng.normal(84,16,2200)}
fig,(a,b)=plt.subplots(1,2,figsize=(10.8,4.5))
bins=np.linspace(0,np.quantile(loss,.995),42); a.hist(loss,bins=bins,density=True,color="#9EC9E2",edgecolor="white")
a.axvline(var,color=OI[1],lw=2,label=f"VaR（{alpha:.0%}）={var:.1f}"); a.axvspan(var,bins[-1],color=OI[3],alpha=.18,label=f"CVaR={cvar:.1f}")
a.set(xlabel="情景损失",ylabel="概率密度"); a.legend()
thresholds=np.linspace(60,115,70)
for (name,values),color in zip(schemes.items(),OI): b.plot(thresholds,[(values>t).mean() for t in thresholds],lw=2,color=color,label=name)
b.axhline(.10,color="#555555",ls="--",lw=1.2,label="允许违反概率 10%")
b.set(xlabel="约束阈值",ylabel="违反概率",ylim=(0,1)); b.legend()
fig.savefig("尾部风险与机会约束.png"); plt.close(fig)
print("已输出：尾部风险与机会约束.png")
```
