## 图表说明

变点分析的完整交代：序列本身、变点在哪（虚线）、每段平均水平（横线+数值）、以及每段内部的增长节奏（箱线）——一段均值高但增长放缓和一段均值低但加速，是两种完全不同的故事。适合政策/产品节点前后的结构性判断。注意：变点算法的惩罚系数决定段数，须写进说明；变点日期表述为"变点后一周首日"之类的口径要统一。

```python
# -*- coding: utf-8 -*-
# 图表模板：时序变点检测双面板（PELT+rbf 分段均值 + 分段增长箱线）
# 数据契约：一条等间隔时序；需 ruptures 库；pen 按模型选择准则校准
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
import ruptures as rpt

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT, C_GREY = "#0072B2", "#D55E00", "#6B6B6B"

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(13)
idx = pd.date_range("2024-01-07", periods=104, freq="W-SUN")
level = np.concatenate([np.full(38, 1.0), np.full(30, 1.45), np.full(36, 1.25)])
season = 1 + 0.15 * np.sin((np.arange(104) - 12) / 52 * 2 * np.pi)
y = 2.6e6 * level * season * rng.lognormal(0, 0.05, 104)

# ===== 2. 变点检测（对数变换后 PELT+rbf；pen 按模型选择准则校准） =====
cps = rpt.Pelt(model="rbf", min_size=8, jump=2).fit(np.log(y).reshape(-1, 1)).predict(pen=6)[:-1]

# ===== 3. 绘图 =====
fig, axes = plt.subplots(2, 1, figsize=(11, 6.2), height_ratios=[2.4, 1],
                         sharex=False, gridspec_kw={"hspace": 0.32})
ax = axes[0]
x = np.arange(len(y))
ax.plot(x, y / 1e6, color=C_MAIN, lw=1.7)
bounds = [0] + cps + [len(y)]
for i, (a, b) in enumerate(zip(bounds[:-1], bounds[1:])):
    mu = y[a:b].mean() / 1e6
    ax.hlines(mu, a, b - 1, color=C_ACCENT, lw=2.4, alpha=0.9)
    ax.text((a + b - 1) / 2, mu + 0.06, f"{mu:.2f}M", ha="center", fontsize=9, color=C_ACCENT)
for c in cps:
    ax.axvline(c - 0.5, color=C_ACCENT, ls="--", lw=1.0, alpha=0.65)
ax.set_ylabel("周期强度（百万）")
ax.set_xticks(x[::8], [d.strftime("%y-%m") for d in idx[::8]], fontsize=8.5)

ax = axes[1]
growth, labels = [], []
for a, b in zip(bounds[:-1], bounds[1:]):
    seg = y[a:b]
    if len(seg) > 2:
        growth.append(np.diff(seg) / seg[:-1] * 100)
        labels.append(f"段{len(labels) + 1}")
bp = ax.boxplot(growth, patch_artist=True, widths=0.5,
                medianprops=dict(color="white"))
for patch in bp["boxes"]:
    patch.set_facecolor(C_MAIN)
    patch.set_alpha(0.75)
ax.axhline(0, color=C_GREY, lw=0.8)
ax.set_xticklabels(labels)
ax.set_ylabel("环比增长（%）")

# ===== 4. 输出（文件名必须与笔记同名） =====
fig.savefig("变点检测双面板.png")
plt.close(fig)
print(f"已输出：变点检测双面板.png；检出 {len(cps)} 个变点")
```
