## 图表说明

上方面板保留时域波形，左下给出全局功率谱并标注主峰，右下用短时傅里叶变换显示频率何时出现。频率解释必须同时报告采样频率、窗长和频率分辨率；非等间隔序列不能直接套用 FFT。

```python
# -*- coding: utf-8 -*-
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.2,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#DDDDDD","grid.linewidth":.6,"axes.axisbelow":True,"savefig.dpi":300,"savefig.bbox":"tight"})
rng=np.random.default_rng(8); fs=50.; t=np.arange(0,24,1/fs)
x=1.15*np.sin(2*np.pi*1.3*t)+.55*np.sin(2*np.pi*4.2*t+.5)+(.75*(t>12))*np.sin(2*np.pi*7.4*t)+rng.normal(0,.28,len(t))

fig=plt.figure(figsize=(10.8,7.1)); gs=fig.add_gridspec(2,2,height_ratios=[.82,1],hspace=.34,wspace=.27)
at=fig.add_subplot(gs[0,:]); at.plot(t,x,color="#34495E",lw=.85); at.axvspan(12,24,color="#E69F00",alpha=.10,label="新增瞬态频率区间"); at.set(xlabel="时间 / s",ylabel="振幅",title="时域信号与结构变化"); at.legend(loc="upper right",frameon=False)

win=np.hanning(len(x)); freq=np.fft.rfftfreq(len(x),1/fs); power=np.abs(np.fft.rfft((x-x.mean())*win))**2; power/=power.max()
ap=fig.add_subplot(gs[1,0]); ap.plot(freq,power,color="#0072B2",lw=1.4); ids=np.argsort(power[1:])[-6:]+1; chosen=[]
for i in ids[np.argsort(power[ids])[::-1]]:
    if all(abs(freq[i]-freq[j])>.45 for j in chosen): chosen.append(i)
    if len(chosen)==3: break
for i in sorted(chosen):
    ap.scatter(freq[i],power[i],color="#D55E00",zorder=3); ap.annotate(f"{freq[i]:.2f} Hz",(freq[i],power[i]),xytext=(5,8),textcoords="offset points",fontsize=9)
ap.set_xlim(0,12); ap.set(xlabel="频率 / Hz",ylabel="归一化功率",title="全局频谱与主峰")

nper=200; step=25; starts=np.arange(0,len(x)-nper+1,step); S=[]
for s in starts:
    seg=(x[s:s+nper]-x[s:s+nper].mean())*np.hanning(nper); S.append(np.abs(np.fft.rfft(seg))**2)
S=np.array(S).T; ff=np.fft.rfftfreq(nper,1/fs); tt=(starts+nper/2)/fs
aspec=fig.add_subplot(gs[1,1]); im=aspec.pcolormesh(tt,ff,10*np.log10(S+1e-8),shading="auto",cmap="magma"); aspec.set_ylim(0,12); aspec.set(xlabel="时间 / s",ylabel="频率 / Hz",title="短时频谱：频率何时出现"); cb=fig.colorbar(im,ax=aspec,pad=.02); cb.set_label("功率 / dB")
fig.savefig("时域频域峰值诊断.png"); plt.close(fig)
print("已输出：时域频域峰值诊断.png")
```
