## 图表说明

自绘的三列桑基：左列到右列的流带用 S 形平滑多边形近似，节点为带数值标签的色条。适合"两类主体经过中间状态落到若干去处"的流量故事（用户×时段×区域、渠道×阶段×结果）。注意：各列总量必须一致（守恒），否则带子对不齐；节点按显示顺序自上而下排列，流值较大的带会优先靠上。

```python
# -*- coding: utf-8 -*-
# 图表模板：三列桑基图（matplotlib 手绘流带）
# 数据契约：三列节点(名称,数值)且列间总量一致；流=(源索引,目标索引)→数值
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.patheffects as pe
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
users = [("会员", 3.1e6), ("散客", 1.9e6)]
slots = [("早峰 6–9 时", 1.4e6), ("午间 11–15 时", 1.1e6),
         ("晚峰 16–19 时", 1.6e6), ("其他时段", 0.9e6)]
regs = [("曼哈顿核心", 2.2e6), ("布鲁克林", 1.2e6), ("皇后区", 0.9e6), ("泽西城/其他", 0.7e6)]
# 用户→时段（行=用户索引，列=时段索引）
f0 = {(0, 0): 1.05e6, (0, 1): 0.72e6, (0, 2): 0.98e6, (0, 3): 0.35e6,
      (1, 0): 0.35e6, (1, 1): 0.38e6, (1, 2): 0.62e6, (1, 3): 0.55e6}
# 时段→区域
f1 = {(0, 0): 0.62e6, (0, 1): 0.35e6, (0, 2): 0.26e6, (0, 3): 0.17e6,
      (1, 0): 0.48e6, (1, 1): 0.28e6, (1, 2): 0.20e6, (1, 3): 0.14e6,
      (2, 0): 0.72e6, (2, 1): 0.38e6, (2, 2): 0.28e6, (2, 3): 0.22e6,
      (3, 0): 0.38e6, (3, 1): 0.19e6, (3, 2): 0.16e6, (3, 3): 0.17e6}
node_colors = [[OI[0], OI[1]],
               [OI[5], OI[2], OI[0], OI[7]],
               [OI[0]] * len(regs)]

# ===== 2. 绘图 =====
def draw_sankey(ax, cols, flows, colors):
    """cols: 每列节点 [(标签, 数值)]; flows: {列i: {(源索引, 目标索引): 数值}}; 列间总量守恒。"""
    ncol = len(cols)
    xs = np.linspace(0.06, 1 - 0.26 - 0.06, ncol)
    bw = 0.028
    ymax = max(sum(v for _, v in c) for c in cols) * 1.12
    node_y = []
    for c in cols:
        y0, pos = ymax * 0.96, []
        for _, v in c:
            pos.append((y0 - v, y0))
            y0 -= v + ymax * 0.028
        node_y.append(pos)
    for ci in range(ncol - 1):                       # 先画流带
        # 偏移必须按节点累计；若按每条边累计，同一节点的流带会彼此重叠。
        src_off = {i: 0.0 for i in range(len(cols[ci]))}
        dst_off = {i: 0.0 for i in range(len(cols[ci + 1]))}
        for (si, di), v in sorted(flows[ci].items(), key=lambda kv: -kv[1]):
            x0, x1 = xs[ci] + bw, xs[ci + 1]
            sy0 = node_y[ci][si][0] + src_off[si]
            dy0 = node_y[ci + 1][di][0] + dst_off[di]
            src_off[si] += v
            dst_off[di] += v
            tt = np.linspace(0, 1, 40)
            sm = 3 * tt**2 - 2 * tt**3               # smoothstep 平滑
            cxs = x0 + (x1 - x0) * sm
            ax.fill(np.concatenate([cxs, cxs[::-1]]),
                    np.concatenate([(sy0 + (dy0 - sy0) * sm),
                                    (sy0 + v + (dy0 + v - sy0 - v) * sm)[::-1]]),
                    color=colors[ci][si], alpha=0.32, lw=0)
    for ci, c in enumerate(cols):                    # 再画节点条与标签
        for ni, ((label, v), (y1, y0)) in enumerate(zip(c, node_y[ci])):
            ax.add_patch(plt.Rectangle((xs[ci], y1), bw, y0 - y1,
                                       fc=colors[ci][ni], ec="none"))
            side = "left" if ci == 0 else "right"
            tx = xs[ci] - 0.012 if side == "left" else xs[ci] + bw + 0.012
            ax.text(tx, (y0 + y1) / 2, f"{label}  {v/1e6:.1f}M",
                    ha="right" if side == "left" else "left", va="center", fontsize=8.5,
                    path_effects=[pe.withStroke(linewidth=2.5, foreground="white")])
    ax.set_xlim(-0.03, 1.18)
    ax.set_ylim(0, ymax)
    ax.axis("off")

fig, ax = plt.subplots(figsize=(11, 6))
draw_sankey(ax, [users, slots, regs], [f0, f1], node_colors)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("三列桑基图.png")
plt.close(fig)
print("已输出：三列桑基图.png")
```
