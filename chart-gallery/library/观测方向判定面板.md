## 图表说明

观测因果发现（CCM/RCM）批量跑完之后的"判定看板"：多少对是单向/双向/不支持（分类条形）、收敛性证据集中在哪个区域（散点四象限）、最终多少对通过晋级门（横条）——三张小图合成一张交代方法严谨性。适合方法章与附录。注意：右下角必须保留"仅观测方向、不等同于 do 效应"的声明；分类类别固定四类，别混入未定类。

```python
# -*- coding: utf-8 -*-
# 图表模板：观测方向判定三面板（分类计数 / 收敛散点 / 门控横条）
# 数据契约：分层×四类计数表 + 逐对收敛指标 + 各门通过数
# 协议：全中文标注；图内不加"XX图"标题；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})

# ===== 1. 合成示例数据（使用时替换为真实判定输出） =====
rng = np.random.default_rng(30)
classes = ["仅甲领先乙", "仅乙领先甲", "双向", "均不支持"]
cc = ["#0f766e", "#2563eb", "#f59e0b", "#94a3b8"]
tiers = ["确认性", "筛查性"]
pivot = np.array([[18, 22, 25, 11], [64, 58, 40, 53]])       # 2 层 × 4 类
slope = rng.normal(0.06, 0.05, 60)
delta = rng.normal(0.02, 0.04, 60)
tier_flag = rng.random(60) < 0.3
gates = [28, 19, 14]

# ===== 2. 绘图 =====
fig = plt.figure(figsize=(13.0, 7.6))
gs = fig.add_gridspec(2, 2, height_ratios=[1.05, 1], wspace=0.30, hspace=0.50)
ax = fig.add_subplot(gs[0, :])
x = np.arange(len(tiers))
width = 0.18
for i, (cls, color) in enumerate(zip(classes, cc)):
    ax.bar(x + (i - 1.5) * width, pivot[:, i], width, label=cls, color=color)
    for xi, v in zip(x + (i - 1.5) * width, pivot[:, i]):
        ax.text(xi, v + 0.8, f"{v:.0f}", ha="center", fontsize=8)
ax.set_xticks(x, tiers)
ax.set_ylabel("成对任务数")
ax.legend(ncol=2, loc="upper left", bbox_to_anchor=(0, 1.04), borderaxespad=0.0)

ax2 = fig.add_subplot(gs[1, 0])
for flag, color, lbl in [(True, "#f59e0b", "筛查性"), (False, "#2563eb", "确认性")]:
    ax2.scatter(slope[tier_flag == flag], delta[tier_flag == flag], s=20,
                alpha=0.55, label=lbl, color=color)
ax2.axhline(0, color="#64748b", linewidth=0.8)
ax2.axvline(0, color="#64748b", linewidth=0.8)
ax2.set_xlabel("收敛斜率")
ax2.set_ylabel("收敛差值")
ax2.legend(frameon=False)

ax3 = fig.add_subplot(gs[1, 1])
ax3.barh(np.arange(3), gates, color=["#64748b", "#0f766e", "#7c3aed"])
ax3.set_yticks(np.arange(3), ["核心门通过", "敏感性门通过", "晋级观测方向"])
ax3.set_xlabel("方向任务数")
for i, v in enumerate(gates):
    ax3.text(v + max(gates) * 0.02, i, f"{v:.0f}", va="center", fontsize=8.5)
ax3.text(0.03, 0.08, "仅为观测动力方向；不等同于 do 效应", transform=ax3.transAxes,
         fontsize=8.5, color="#991b1b", va="bottom")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("观测方向判定面板.png")
plt.close(fig)
print("已输出：观测方向判定面板.png")
```
