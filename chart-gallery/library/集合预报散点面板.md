## 图表说明

同一预报对象在多个时效上叠加集合成员、均值、真值和经验覆盖椭圆。椭圆概括二维离散方向，真值是否落入椭圆提供直观但非正式的覆盖诊断。各面板坐标必须一致；云的发散应来自真实初始化、参数或随机扰动。椭圆依赖近似椭圆分布，明显多峰时应改画分位数等高线。

```python
# -*- coding: utf-8 -*-
# 图表模板：集合预报散点面板（单值+集合云+均值+真值，2×2 分时效）
# 数据契约：真值轨迹 + 扰动初始化的集合轨迹，按时效取点
# 协议：全中文标注；图内不加"XX图"标题；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from matplotlib.patches import Ellipse

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})

# ===== 1. 合成示例数据：Hénon 映射真值 + 扰动初始化集合 =====
def henon_traj(x0, y0, n, a=1.4, b=0.3):
    xs, ys = [x0], [y0]
    for _ in range(n):
        x0, y0 = 1 - a * x0 * x0 + y0, b * x0
        xs.append(x0)
        ys.append(y0)
    return np.array(xs), np.array(ys)

rng = np.random.default_rng(11)
N = 60
tx, ty = henon_traj(0.1, 0.1, N)                       # 真值轨迹
ens = []
for _ in range(240):                                   # 240 条集合轨迹（扰动初始化）
    ex0 = tx[0] + rng.normal(0, 0.004)
    ey0 = ty[0] + rng.normal(0, 0.004)
    ens.append(henon_traj(ex0, ey0, N))
leads = [1, 6, 23, 28]

# ===== 2. 绘图 =====
fig, axes = plt.subplots(2, 2, figsize=(11.5, 9), sharex=True, sharey=True)
for ax, k in zip(axes.flat, leads):
    exs = np.array([e[0][k] for e in ens])
    eys = np.array([e[1][k] for e in ens])
    ax.scatter(exs, eys, s=9, color="#7F8C99", alpha=0.32, edgecolors="none", label="集合成员")
    cov = np.cov(np.vstack([exs, eys]))
    eigval, eigvec = np.linalg.eigh(cov)
    order = eigval.argsort()[::-1]
    eigval, eigvec = eigval[order], eigvec[:, order]
    angle = np.degrees(np.arctan2(eigvec[1, 0], eigvec[0, 0]))
    scale80 = np.sqrt(3.219)                            # 二维正态近似下 80% 椭圆
    ellipse = Ellipse(
        (exs.mean(), eys.mean()),
        width=2 * scale80 * np.sqrt(eigval[0]),
        height=2 * scale80 * np.sqrt(eigval[1]),
        angle=angle, facecolor="#56B4E9", edgecolor="#0072B2",
        alpha=0.13, lw=1.2, zorder=2,
    )
    ax.add_patch(ellipse)
    ax.scatter(tx[k], ty[k], marker="*", s=190, color="#D55E00", zorder=5, label="真值")
    ax.scatter(exs.mean(), eys.mean(), marker="D", s=48, color="#0072B2",
               zorder=4, label="集合均值")
    delta = np.array([tx[k] - exs.mean(), ty[k] - eys.mean()])
    inside = float(delta @ np.linalg.pinv(cov) @ delta) <= 3.219
    ax.set_title(f"第 {k} 步预报｜真值{'在' if inside else '不在'}80%椭圆内", fontsize=10.0)
    ax.set_xlabel("状态 x")
    ax.set_ylabel("状态 y")
handles = [plt.Line2D([], [], marker="*", ls="", color="#D55E00", ms=13, label="真值"),
           plt.Line2D([], [], marker="D", ls="", color="#0072B2", ms=8, label="集合均值"),
           plt.Line2D([], [], marker="o", ls="", color="#7F8C99", ms=6, label="集合成员"),
           mpl.patches.Patch(facecolor="#56B4E9", edgecolor="#0072B2", alpha=0.3, label="80%覆盖椭圆")]
axes[0, 0].legend(handles=handles, loc="upper left")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("集合预报散点面板.png")
plt.close(fig)
print("已输出：集合预报散点面板.png")
```
