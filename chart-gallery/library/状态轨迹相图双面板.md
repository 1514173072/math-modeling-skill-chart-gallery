## 图表说明

左侧判断振幅、周期和暂态过程，右侧观察轨道是否收敛到稳定点或极限环。相图必须使用同步状态变量；把两个无关时间序列作散点图不能解释为动力系统相轨迹。

```python
# -*- coding: utf-8 -*-
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from matplotlib.collections import LineCollection

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.3,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#DDDDDD","grid.linewidth":.6,"axes.axisbelow":True,"savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
mu=1.7; dt=.015; t=np.arange(0,30,dt); state=np.zeros((len(t),2)); state[0]=[2.6,0.15]
def f(z): x,v=z; return np.array([v,mu*(1-x*x)*v-x])
for i in range(len(t)-1):
    z=state[i]; k1=f(z); k2=f(z+dt*k1/2); k3=f(z+dt*k2/2); k4=f(z+dt*k3); state[i+1]=z+dt*(k1+2*k2+2*k3+k4)/6
x,v=state.T

fig,(at,ap)=plt.subplots(1,2,figsize=(11.0,5.7),gridspec_kw={"width_ratios":[1.05,1]})
at.plot(t,x,color="#0072B2",lw=1.2,label="状态 x(t)"); at.plot(t,v,color="#D55E00",lw=1.0,alpha=.85,label="状态 v(t)"); at.axvspan(0,8,color="#999999",alpha=.10,label="暂态区间"); at.set(xlabel="时间",ylabel="状态值",title="状态轨迹：暂态进入稳定振荡"); at.legend(loc="upper right")

gx=np.linspace(-3,3,19); gv=np.linspace(-4,4,19); X,V=np.meshgrid(gx,gv); U=V; W=mu*(1-X*X)*V-X; speed=np.hypot(U,W); ap.streamplot(X,V,U,W,color=np.log1p(speed),cmap="Greys",density=.75,linewidth=.55,arrowsize=.65)
pts=np.c_[x,v].reshape(-1,1,2); seg=np.concatenate([pts[:-1],pts[1:]],axis=1); lc=LineCollection(seg,cmap="viridis",norm=plt.Normalize(t.min(),t.max()),linewidth=1.8); lc.set_array(t[:-1]); ap.add_collection(lc)
ap.scatter(x[0],v[0],s=45,color="#E69F00",edgecolor="white",zorder=4,label="起点"); ap.scatter(x[-1],v[-1],s=45,color="#009E73",edgecolor="white",zorder=4,label="终点"); ap.set_xlim(-3,3); ap.set_ylim(-4,4); ap.set(xlabel="状态 x",ylabel="状态 v",title="相图：轨道方向与极限环"); ap.legend(loc="upper right"); cb=fig.colorbar(lc,ax=ap,pad=.02); cb.set_label("时间")
fig.subplots_adjust(wspace=.28); fig.savefig("状态轨迹相图双面板.png"); plt.close(fig)
print("已输出：状态轨迹相图双面板.png")
```
