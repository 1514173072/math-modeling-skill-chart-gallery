## 图表说明

左图越接近对角线表示概率越可信，右图用于判断偏差是否来自样本稀疏区。校准好不等于区分能力强，必要时应与 ROC/PR 或 Brier 分数共同报告；分箱点必须按箱内真实频率计算，不能用平滑曲线替代原始证据。

```python
# -*- coding: utf-8 -*-
# 数据契约：二元 y_true + 各模型预测概率
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from sklearn.calibration import calibration_curve
from sklearn.metrics import brier_score_loss

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#D9D9D9","savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73","#D55E00","#CC79A7","#56B4E9","#F0E442","#999999"]
rng=np.random.default_rng(7); n=900
latent=rng.normal(size=n); true_p=1/(1+np.exp(-latent)); y=rng.binomial(1,true_p)
pred={"模型甲":np.clip(true_p*.84+.08,0,1),"模型乙":np.clip(true_p**1.55,0,1)}
fig,(ax,axh)=plt.subplots(1,2,figsize=(10.3,4.5),gridspec_kw={"width_ratios":[1,1.15]})
ax.plot([0,1],[0,1],"--",color="#777777",lw=1.2,label="理想校准")
for (name,p),color in zip(pred.items(),OI):
    obs,mean=calibration_curve(y,p,n_bins=10,strategy="quantile")
    brier=brier_score_loss(y,p)
    ax.plot(mean,obs,"-o",ms=4,lw=1.8,color=color,label=f"{name}（Brier={brier:.3f}）")
    axh.hist(p,bins=np.linspace(0,1,16),histtype="step",lw=1.8,color=color,label=name)
ax.set(xlim=(0,1),ylim=(0,1),xlabel="平均预测概率",ylabel="实际发生率"); ax.legend(loc="upper left")
axh.set(xlabel="预测概率",ylabel="样本数"); axh.legend()
fig.savefig("可靠性校准双面板.png"); plt.close(fig)
print("已输出：可靠性校准双面板.png")
```
