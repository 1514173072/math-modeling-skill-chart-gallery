## 图表说明

两组在若干类别（如频段、年龄段、渠道）上的估计结果用横向点区间表达。点表示均值或效应估计，横线表示预先定义的置信区间，样本量直接标在右侧。它比柱状图更适合比较估计位置和不确定性。置信区间的计算方法必须在正文说明，不能把标准差、标准误和置信区间混用。

```python
# -*- coding: utf-8 -*-
# 图表模板：分组点区间图（组间×多类别）
# 数据契约：类别、分组、估计值、区间下限、区间上限、样本量
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实估计与区间） =====
rng = np.random.default_rng(15)
cats = ["频段一", "频段二", "频段三", "频段四", "频段五", "频段六"]
est_a = np.array([3.2, 4.1, 2.6, 1.9, 1.4, 1.1])
est_b = np.array([2.5, 2.8, 2.2, 2.0, 1.8, 1.6])
half_a = rng.uniform(0.22, 0.48, len(cats))
half_b = rng.uniform(0.22, 0.48, len(cats))
n_a = np.array([82, 77, 91, 68, 74, 63])
n_b = np.array([76, 81, 85, 72, 69, 66])

# ===== 2. 绘图 =====
y = np.arange(len(cats))
fig, ax = plt.subplots(figsize=(9, 5.2))
for offset, est, half, ns, color, label in [
    (-0.14, est_a, half_a, n_a, OI[0], "组甲"),
    (0.14, est_b, half_b, n_b, OI[1], "组乙"),
]:
    ax.errorbar(est, y + offset, xerr=half, fmt="o", ms=5.5, capsize=3,
                color=color, ecolor=color, lw=1.4, label=label)
    for x0, y0, n in zip(est + half, y + offset, ns):
        ax.text(x0 + 0.08, y0, f"n={n}", va="center", fontsize=8.2, color="#555555")
ax.set_yticks(y, cats)
ax.invert_yaxis()
ax.set_xlabel("估计值（点）与 95% 置信区间（横线）")
ax.set_ylabel("类别")
ax.margins(x=0.12)
ax.legend(loc="upper right")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("误差棒分组柱状图.png")
plt.close(fig)
print("已输出：误差棒分组柱状图.png")
```
