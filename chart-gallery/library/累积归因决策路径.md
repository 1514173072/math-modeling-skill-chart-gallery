## 图表说明

每条线从共同基线出发，依次叠加各特征贡献后到达最终预测。图只适用于满足加性分解的数据，例如 SHAP 值；特征顺序必须固定并说明排序依据，贡献方向不能被解释为因果效应。

```python
# -*- coding: utf-8 -*-
# 数据契约：样本×特征的加性贡献矩阵 + 基线；终点=基线+贡献和
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#E1E1E1","savefig.dpi":300,"savefig.bbox":"tight"})
rng=np.random.default_rng(31); features=["资源水平","服务半径","人口密度","交通便利","需求波动","政策支持"]
base=.42; contrib=rng.normal([.08,-.03,.12,.05,-.07,.04],[.05,.04,.08,.05,.05,.04],(24,6))
order=np.argsort(np.mean(np.abs(contrib),axis=0)); contrib=contrib[:,order]; features=[features[i] for i in order]
paths=np.c_[np.full(len(contrib),base),base+np.cumsum(contrib,axis=1)]
final=paths[:,-1]; norm=mpl.colors.Normalize(final.min(),final.max()); cmap=plt.cm.coolwarm
fig,ax=plt.subplots(figsize=(8.1,5.5))
for row,value in zip(paths,final): ax.plot(range(7),row,color=cmap(norm(value)),alpha=.58,lw=1.2)
ax.scatter(np.full(len(final),6),final,c=final,cmap=cmap,norm=norm,s=20,zorder=3)
ax.axhline(base,color="#555555",ls="--",lw=1,label=f"基线 {base:.2f}")
ax.set_xticks(range(7),["基线"]+features,rotation=28,ha="right"); ax.set_ylabel("累积模型输出"); ax.legend(loc="upper left")
cb=fig.colorbar(mpl.cm.ScalarMappable(norm=norm,cmap=cmap),ax=ax,pad=.02); cb.set_label("最终预测")
fig.savefig("累积归因决策路径.png"); plt.close(fig)
print("已输出：累积归因决策路径.png")
```
