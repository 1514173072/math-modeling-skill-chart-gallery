## 图表说明

讲"嵌入空间学到了什么"的标准画法：挑 3 条形态各异的序列，从序列末端拉引线到右图嵌入空间中对应的点——同一簇的序列形态相近这一事实，不需要任何量化指标就有了直观证据。适合 UMAP/t-SNE/PCA 降维结果的解读。注意：嵌入坐标离线算好后随图给出（本模板用合成聚类坐标代替 UMAP，用法一致）；引线只锚 2–4 条，多了会乱；背景序列压灰降低视觉权重。

```python
# -*- coding: utf-8 -*-
# 图表模板：时序-嵌入空间联动图（引线锚定）
# 数据契约：重点时序+背景时序；二维嵌入坐标离线算好
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from matplotlib.patches import ConnectionPatch

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73"]

# ===== 1. 合成示例数据（嵌入坐标真实使用时由 UMAP/t-SNE 离线算出） =====
rng = np.random.default_rng(13)
t = np.arange(120)
series, kinds = [], []
for _ in range(40):                                    # 背景序列（索引 0–39）
    series.append(rng.normal(0, 1, 120).cumsum() * 0.02)
    kinds.append("背景")
hl = [("周期型", 3 * np.sin(t / 12) + rng.normal(0, 0.15, 120), 0),
      ("趋势型", t / 60 + rng.normal(0, 0.2, 120), 1),
      ("突变型", np.where(t > 70, 2.2, 0) + rng.normal(0, 0.2, 120), 2)]
for name, s, ci in hl:                                 # 重点序列（索引 40–42）
    series.append(s)
    kinds.append((name, ci))
cluster_of = {40: 0, 41: 1, 42: 2}                     # 重点序列索引 → 所属簇
embed = np.vstack([rng.normal(loc, 0.22, (n, 2)) for loc, n in
                   [((-1.2, 0.6), 14), ((1.6, 0.8), 13), ((0.2, -1.1), 13)]])

# ===== 2. 绘图 =====
fig = plt.figure(figsize=(12.5, 4.9))
gs = fig.add_gridspec(1, 2, width_ratios=[1.15, 1], wspace=0.22)
ax1 = fig.add_subplot(gs[0, 0])
ax2 = fig.add_subplot(gs[0, 1])
for k, (s, kind) in enumerate(zip(series, kinds)):
    if kind == "背景":
        ax1.plot(t, s, color="#C7D0D9", lw=0.7, alpha=0.55)
        continue
    name, ci = kind
    cc = OI[ci]
    ax1.plot(t, s, color=cc, lw=1.8)
    ax1.annotate(name, xy=(118, s[-1]), xytext=(122, s[-1]),
                 fontsize=9.5, color=cc, va="center")
ax1.set_xlim(0, 138)
ax1.set_xlabel("时间步")
ax1.set_ylabel("归一化强度")

emb_lab = [0] * 14 + [1] * 13 + [2] * 13
for ci, cc in enumerate(OI):
    pts = embed[np.array(emb_lab) == ci]
    ax2.scatter(pts[:, 0], pts[:, 1], s=16, color=cc, alpha=0.5,
                edgecolors="none", label=f"簇 {ci + 1}")
ax2.set_xlabel("嵌入维度一")
ax2.set_ylabel("嵌入维度二")
ax2.legend(loc="lower right")

anchors = {ci: embed[np.array(emb_lab) == ci][np.argmin(
    embed[np.array(emb_lab) == ci][:, 0])] for ci in range(3)}   # 取各簇最左点，引线从左侧进入
for k, ci in cluster_of.items():                       # 引线：序列末端 → 嵌入锚点
    s = series[k]
    cc = OI[ci]
    ex, ey = anchors[ci]
    con = ConnectionPatch(xyA=(118, s[-1]), coordsA=ax1.transData,
                          xyB=(ex, ey), coordsB=ax2.transData,
                          color=cc, lw=0.9, ls=":", alpha=0.6,
                          connectionstyle="arc3,rad=-0.08")
    fig.add_artist(con)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("嵌入联动图.png")
plt.close(fig)
print("已输出：嵌入联动图.png")
```
