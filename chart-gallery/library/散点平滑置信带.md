响应关系的第一张图：原始散点保留全部信息，核平滑给出趋势，成对 bootstrap 区间表达曲线估计的不确定性。注意：这是描述性可视化，不是效应估计；带宽需在分析前固定或用交叉验证选择，边界区域样本过少时不作强解释。

```python
# -*- coding: utf-8 -*-
# 图表模板：散点+核平滑置信带（分组描述性关系）
# 数据契约：自变量、响应两列数值 + 二分条件列；bootstrap 成对重采样观测行
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT = "#0072B2", "#D55E00"

# ===== 1. 工具函数（通用，可直接复用） =====
def kernel_smooth(x, y, grid, bandwidth):
    """Nadaraya-Watson 核平滑；只承担描述性趋势展示。"""
    z = (x[:, None] - grid[None, :]) / bandwidth
    w = np.exp(-0.5 * z**2)
    return (w * y[:, None]).sum(axis=0) / w.sum(axis=0)


def smooth(x, y, n_grid=120, n_boot=300, seed=7):
    """成对 bootstrap：每次同时重采样 x 与 y，保持观测配对。"""
    rng = np.random.default_rng(seed)
    lo_x, hi_x = np.quantile(x, [0.03, 0.97])
    grid = np.linspace(lo_x, hi_x, n_grid)
    bandwidth = max((hi_x - lo_x) / 10, np.finfo(float).eps)
    yhat = kernel_smooth(x, y, grid, bandwidth)
    boots = []
    for _ in range(n_boot):
        idx = rng.integers(0, len(x), len(x))
        boots.append(kernel_smooth(x[idx], y[idx], grid, bandwidth))
    boots = np.vstack(boots)
    lo = np.nanpercentile(boots, 2.5, axis=0)
    hi = np.nanpercentile(boots, 97.5, axis=0)
    return grid, yhat, lo, hi

# ===== 2. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(3)
n = 500
df = pd.DataFrame({
    "x": rng.normal(15, 9, n).clip(-12, 34),                 # 自变量（如气温）
    "cond": rng.random(n) < 0.28,                            # 二分条件（如有无干预）
})
sat = 95 / (1 + np.exp(-(df["x"] - 16) / 6))
df["y"] = sat * np.where(df["cond"], 0.72, 1.0) * rng.lognormal(0, 0.08, n)

# ===== 3. 绘图 =====
fig, ax = plt.subplots(figsize=(9, 4.6))
for mask, color, label in [(~df["cond"], C_MAIN, "条件甲"), (df["cond"], C_ACCENT, "条件乙")]:
    d = df[mask]
    ax.scatter(d["x"], d["y"], s=9, alpha=0.35, color=color, edgecolors="none")
    xs, yhat, lo, hi = smooth(d["x"].to_numpy(), d["y"].to_numpy())
    ax.plot(xs, yhat, color=color, lw=2.4, label=label)
    ax.fill_between(xs, lo, hi, color=color, alpha=0.13)

ax.set_xlabel("自变量")
ax.set_ylabel("响应")
ax.legend(loc="upper left")

# ===== 4. 输出（文件名必须与笔记同名） =====
fig.savefig("散点平滑置信带.png")
plt.close(fig)
print("已输出：散点平滑置信带.png")
```
