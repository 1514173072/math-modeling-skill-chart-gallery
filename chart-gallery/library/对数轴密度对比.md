## 图表说明

右偏长尾数据（时长/金额/时延）线性轴会被尾部压扁，对数轴 + 阶梯密度才是它的母语；两组各一条阶梯线加浅填充，中位数竖线直接标数，位置差即分布差。适合服务时长、订单金额、响应时间等运营指标的群体对比。注意：截断（如 >60 分钟）要在轴标签写明；对数轴刻度手动设 1/2/5/10/20/60 这类可读数，不要用默认指数。

```python
# -*- coding: utf-8 -*-
# 图表模板：对数轴密度对比（阶梯直方图）
# 数据契约：每组一条正数值序列（长尾右偏），对数分箱，组数 2–4
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(10)
grp_a = np.clip(rng.lognormal(np.log(11), 0.55, 4000), 1, 60)   # 短而集中
grp_b = np.clip(rng.lognormal(np.log(17), 0.75, 4000), 1, 60)   # 长而散
groups = [(grp_a, OI[0], "组甲"), (grp_b, OI[3], "组乙")]
bins = np.logspace(0, np.log10(60), 48)

# ===== 2. 绘图 =====
fig, ax = plt.subplots(figsize=(8.6, 4.4))
for vals, cc, lbl in groups:
    ax.hist(vals, bins=bins, density=True, histtype="step", lw=2.2, color=cc, label=lbl)
    ax.hist(vals, bins=bins, density=True, color=cc, alpha=0.12)
top = ax.get_ylim()[1]
for vals, cc, lbl in groups:
    med = np.median(vals)
    ax.axvline(med, color=cc, ls="--", lw=1.2, alpha=0.85)
    ax.text(med * 1.06, top * 0.88, f"中位 {med:.1f}",
            fontsize=9, color=cc, rotation=90, va="top")

ax.set_xscale("log")
ax.set_xticks([1, 2, 5, 10, 20, 60], ["1", "2", "5", "10", "20", "60"])
ax.set_xlabel("数值（对数轴，截断至 60）")
ax.set_ylabel("密度")
ax.legend(loc="upper right")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("对数轴密度对比.png")
plt.close(fig)
print("已输出：对数轴密度对比.png")
```
