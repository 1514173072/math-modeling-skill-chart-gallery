## 图表说明

CCM/RCM 类收敛检验的分面升级版：单面板收敛曲线（见《观测方向判定面板》）只回答"这对序列有没有因果"，这张把行=超参一、列=超参二铺成网格，能进一步回答"结论对参数选择稳不稳"——好结果应当整片网格都收敛到同一渐近线，而不是只在个别格子成立。每个子图配 y=参考虚线（如 0.5 判定线与 1.0 理想线），图例收在图顶层只出一次。样本量用对数布点；曲线加小噪声才有"实验感"，别画成光滑理论线。

```python
# -*- coding: utf-8 -*-
# 图表模板：分面收敛曲线（双参数 3×3 网格）
# 数据契约：分面参数（行/列）× 方法曲线 × 样本量网格的收敛得分长表
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG

import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9, "ytick.labelsize": 9,
    "legend.fontsize": 9.5, "axes.titlesize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实收敛得分长表） =====
rng = np.random.default_rng(9)
n_grid = np.geomspace(300, 10000, 8)
a_rows, a_cols = [2, 3, 4], [2, 3, 4]        # 行/列分面参数

def curve(target, speed, n):
    # 从低处起步、随样本量升向渐近线（CCM 技能曲线的标准形态）
    return target * (1 - np.exp(-n / speed)) + rng.normal(0, 0.02, len(n))

# ===== 2. 绘图（3×3 分面 + 图级图例） =====
fig, axes = plt.subplots(3, 3, sharex=True, sharey=True, figsize=(10.2, 7.6))
for i, a1 in enumerate(a_rows):
    for j, a2 in enumerate(a_cols):
        ax = axes[i, j]
        gap = abs(a1 - a2)
        fwd = curve(0.92 - 0.10 * gap, 380 + 160 * gap, n_grid)    # 正向
        rev = curve(0.55 - 0.08 * gap, 620 + 260 * gap, n_grid)    # 反向
        null = rng.normal(0.0, 0.02, len(n_grid))                  # 无因果对照
        ax.plot(n_grid, fwd, "-o", color=OI[0], ms=3.5, lw=1.4)
        ax.plot(n_grid, rev, "-s", color=OI[3], ms=3.5, lw=1.4)
        ax.plot(n_grid, null, "--", color="#555555", lw=1.3)
        ax.axhline(0.5, color="#C8C8C8", ls=":", lw=0.9)
        ax.axhline(1.0, color="#C8C8C8", ls=":", lw=0.9)
        ax.set_title(f"$\\alpha_1$={a1}，$\\alpha_2$={a2}")
        ax.set_xscale("log")
        ax.set_xticks([300, 1000, 3000, 10000], ["300", "1千", "3千", "1万"])
        ax.set_ylim(-0.08, 1.08)
fig.supxlabel("样本量 n（对数刻度）")
fig.supylabel("因果尾系数估计值")
handles = [mpl.lines.Line2D([], [], color=OI[0], marker="o", ms=4, lw=1.4, label="正向估计"),
           mpl.lines.Line2D([], [], color=OI[3], marker="s", ms=4, lw=1.4, label="反向估计"),
           mpl.lines.Line2D([], [], color="#555555", ls="--", lw=1.3, label="无因果对照"),
           mpl.lines.Line2D([], [], color="#C8C8C8", ls=":", lw=0.9, label="参考线 0.5 / 1.0")]
fig.legend(handles=handles, loc="upper center", ncol=4, bbox_to_anchor=(0.5, 1.0))
fig.subplots_adjust(top=0.87, hspace=0.38, wspace=0.18)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("分面收敛曲线.png")
plt.close(fig)
print("已输出：分面收敛曲线.png")
```
