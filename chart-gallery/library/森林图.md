## 图表说明

每个变量一行，点为比率效应，横线为 95% 区间，1 为无效应参考线。横轴使用对数尺度，使 0.5 与 2 这类互为倒数的效应在视觉上距参考线相等。区间不跨 1 只表示在给定模型与区间方法下具有统计证据，不自动等于因果效应或实际重要。所有行必须使用同一种比率口径。

```python
# -*- coding: utf-8 -*-
# 图表模板：效应森林图（IRR + 95% 置信区间）
# 数据契约：行=变量、列=效应量 irr 与区间 lo/hi；同一模型同一参考类
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT = "#0072B2", "#D55E00"

# ===== 1. 合成示例数据（使用时替换为真实回归输出） =====
eff = pd.DataFrame({
    "name": ["条件甲", "条件乙", "干预 A", "干预 B", "季节因子", "连续变量（每 +1 单位）"],
    "irr": [0.88, 0.62, 0.45, 0.83, 0.94, 0.997],
    "lo":  [0.84, 0.55, 0.38, 0.80, 0.90, 0.991],
    "hi":  [0.92, 0.70, 0.53, 0.86, 0.98, 1.004],
})

# ===== 2. 绘图 =====
fig, ax = plt.subplots(figsize=(8.6, 4.4))
y = range(len(eff))
ax.axvline(1.0, color="#999", lw=1.0, ls="--")
ax.errorbar(eff["irr"], y, xerr=[eff["irr"] - eff["lo"], eff["hi"] - eff["irr"]],
            fmt="o", ms=5.5, ecolor="#BBB", elinewidth=1.6, capsize=3, zorder=2)
for i, r in eff.iterrows():
    c = C_ACCENT if r["lo"] > 1 or r["hi"] < 1 else C_MAIN   # 区间跨 1 与否决定染色
    ax.scatter(r["irr"], i, color=c, s=46, zorder=3)
    ax.text(r["hi"], i, f"  {r['irr']:.2f}", va="center", fontsize=9, color="#444")
ax.set_yticks(list(y), eff["name"])
ax.set_xscale("log")
ax.set_xticks([0.4, 0.5, 0.7, 1.0, 1.5])
ax.get_xaxis().set_major_formatter(mpl.ticker.ScalarFormatter())
ax.set_xlabel("效应比 IRR（相对基准，95% 置信区间）")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("森林图.png")
plt.close(fig)
print("已输出：森林图.png")
```
