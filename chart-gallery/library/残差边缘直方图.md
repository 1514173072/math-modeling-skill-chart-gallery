## 图表说明

模型诊断的联合分布版式：主散点回答"残差是否随自变量系统漂移"（弯曲=有结构没抓住），边缘直方图回答"整体精度谁高"（双峰=两类误差混在一起）。适合回归模型的最终诊断图。注意：边缘直方图与主区共享对应坐标范围；两方法用色板区分并在直方图里保持同色；样本大时散点加透明度或抽稀。

```python
# -*- coding: utf-8 -*-
# 图表模板：残差散点 + 边缘直方图（双方法精度对比）
# 数据契约：每方法一组（自变量, 残差）；边缘直方图自动汇总
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实残差） =====
rng = np.random.default_rng(8)
x = rng.uniform(0, 10, 320)
res_a = rng.normal(0, 0.62, 320)                      # 方法甲：同方差
res_b = 0.10 * (x - 5) + rng.normal(0, 0.5, 320)      # 方法乙：随自变量漂移
groups = [("方法甲", x, res_a, OI[0]), ("方法乙", x, res_b, OI[1])]

# ===== 2. 绘图（主区 + 上/右边缘直方图） =====
fig = plt.figure(figsize=(9, 6.4))
gs = fig.add_gridspec(2, 2, width_ratios=[4, 1], height_ratios=[1, 4],
                      wspace=0.04, hspace=0.04)
ax = fig.add_subplot(gs[1, 0])
axt = fig.add_subplot(gs[0, 0], sharex=ax)
axr = fig.add_subplot(gs[1, 1], sharey=ax)
for name, xv, rv, cc in groups:
    ax.scatter(xv, rv, s=10, alpha=0.45, color=cc, edgecolors="none", label=name)
    axt.hist(rv, bins=32, density=True, alpha=0.5, color=cc)
    axr.hist(xv, bins=32, density=True, alpha=0.5, color=cc, orientation="horizontal")
ax.axhline(0, color="#777", lw=0.9, ls="--")
ax.set_xlabel("自变量")
ax.set_ylabel("残差")
ax.legend(markerscale=2.2, loc="upper right")
plt.setp(axt.get_xticklabels(), visible=False)
plt.setp(axr.get_yticklabels(), visible=False)
axt.axis(ymin=0)
axr.axis(xmin=0)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("残差边缘直方图.png")
plt.close(fig)
print("已输出：残差边缘直方图.png")
```
