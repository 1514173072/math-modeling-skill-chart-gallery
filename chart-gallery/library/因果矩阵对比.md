## 图表说明

因果发现结果对比的标准版式：第一个面板是真值，其后每个方法一个面板，蓝格=判定存在因果边；与真值不符的格子（多判/漏判）用红框圈出，方法优劣一眼见分晓。适合 CCM/RCM/PCMI 等任何输出邻接结构的方法评估。注意：行列变量顺序所有面板必须一致；变量名只在最左/最下面板标注一次；方法数多时按 TPR 排序展示。

```python
# -*- coding: utf-8 -*-
# 图表模板：因果矩阵对比图（真值 + 多方法 + 误差红框标注）
# 数据契约：各方法 0/1 邻接矩阵，行列顺序一致；误差格从真值推导
# 协议：全中文标注；图内不加"XX图"标题；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from matplotlib.colors import ListedColormap

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9, "ytick.labelsize": 9,
    "figure.dpi": 110, "savefig.dpi": 300, "savefig.bbox": "tight",
})
CMAP = ListedColormap(["#F2F5F9", "#0072B2"])

# ===== 1. 合成示例数据（使用时替换为真实判定矩阵） =====
rng = np.random.default_rng(6)
n_var = 6
truth = np.zeros((n_var, n_var), dtype=int)
for i, j in [(0, 2), (1, 3), (2, 4), (3, 4), (3, 5), (0, 5)]:
    truth[i, j] = 1
methods = {"基准甲": None, "基准乙": None, "经典法": None, "改进法": None, "本文方法": None}
for name, (fp, fn) in zip(methods, [(2, 2), (3, 1), (1, 2), (2, 0), (1, 0)]):
    m = truth.copy()
    fp_cells = [(rng.integers(n_var), rng.integers(n_var)) for _ in range(fp + 4)]
    fn_cells = [(i, j) for i, j in zip(*np.where(truth == 1))]
    rng.shuffle(fn_cells)
    for _ in range(fp):
        i, j = fp_cells.pop()
        if m[i, j] == 0 and i != j:
            m[i, j] = 1
    for _ in range(fn):
        i, j = fn_cells.pop()
        m[i, j] = 0
    methods[name] = m

# ===== 2. 绘图 =====
names = ["真值"] + list(methods)
mats = [truth] + list(methods.values())
fig, axes = plt.subplots(1, len(names), figsize=(2.05 * len(names), 2.9))
for k, (ax, name, m) in enumerate(zip(axes, names, mats)):
    ax.imshow(m, cmap=CMAP, vmin=0, vmax=1)
    if k > 0:                                   # 误差格红框
        fp = (m == 1) & (truth == 0)
        fn = (m == 0) & (truth == 1)
        for mask in (fp, fn):
            for i, j in zip(*np.where(mask)):
                ax.add_patch(plt.Rectangle((j - 0.5, i - 0.5), 1, 1,
                                           fill=False, ec="#D55E00", lw=1.6))
    ax.set_xticks(range(n_var), [f"变量{chr(65 + i)}" for i in range(n_var)],
                  rotation=90, fontsize=7)
    ax.set_yticks(range(n_var), [f"变量{chr(65 + i)}" for i in range(n_var)], fontsize=7)
    for s in ax.spines.values():
        s.set_visible(False)
    ax.set_xticks(np.arange(-0.5, n_var), minor=True)
    ax.set_yticks(np.arange(-0.5, n_var), minor=True)
    ax.grid(which="minor", color="white", lw=1.2)
    ax.tick_params(which="both", length=0)
    ax.set_title(name, fontsize=10)
axes[0].set_ylabel("原因 →")

fig.text(0.5, 0.005, "橙框 = 与真值不符（多判 / 漏检）", ha="center",
         fontsize=9, color="#D55E00")
# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("因果矩阵对比.png")
plt.close(fig)
print("已输出：因果矩阵对比.png")
```
