## 图表说明

浅灰线给出全部候选，彩色粗线突出按预先声明规则选出的代表方案。指标必须同向标准化，成本或风险要先反向；轴顺序按决策逻辑固定，不能为获得更整齐的线形而调整。候选过多时只显示背景分位带和少量入选方案。

```python
# -*- coding: utf-8 -*-
# 数据契约：方案×4–8个0–1同向指标 + 入选标记
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#D9D9D9","savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73","#D55E00","#CC79A7","#56B4E9","#F0E442","#999999"]
rng=np.random.default_rng(18); dims=["收益","稳定性","可实施性","公平性","环境效益","韧性"]
raw=rng.beta(2.4,2.0,(70,len(dims))); raw[:,1]=np.clip(.75-raw[:,0]*.35+rng.normal(0,.13,70),0,1)
weights=np.array([.25,.16,.18,.14,.14,.13]); score=raw@weights
chosen=np.argsort(score)[-4:]
x=np.arange(len(dims)); fig,ax=plt.subplots(figsize=(9.2,5.2))
for row in raw: ax.plot(x,row,color="#B8BEC5",alpha=.18,lw=.8)
q25,q50,q75=np.quantile(raw,[.25,.5,.75],axis=0); ax.fill_between(x,q25,q75,color="#7F8C8D",alpha=.14,label="候选方案四分位区间"); ax.plot(x,q50,color="#666666",ls="--",lw=1)
for rank,(idx,color) in enumerate(zip(chosen[::-1],OI),1): ax.plot(x,raw[idx],"-o",color=color,lw=2.2,ms=4,label=f"入选{rank}（得分{score[idx]:.2f}）")
ax.set_xticks(x,dims); ax.set_ylim(0,1); ax.set_ylabel("标准化得分（越高越优）"); ax.legend(ncols=2,loc="lower center",bbox_to_anchor=(.5,1.01))
fig.savefig("平行坐标方案筛选.png"); plt.close(fig)
print("已输出：平行坐标方案筛选.png")
```
