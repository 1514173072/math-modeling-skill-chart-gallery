## 图表说明

上面板先用中位数和 10%–90% 区间总结多个起报点的误差幅度，再在少数代表时效显示误差方向。这样避免密集箭头让读者只能看到“很乱”，同时保留偏差方向信息。每根箭头必须从第 k 步真值指向同一起报点的第 k 步集合均值，不能拿预测终点与起报状态比较。

```python
# -*- coding: utf-8 -*-
# 图表模板：误差向量演化图（分时效 quiver：真值 → 集合均值）
# 数据契约：真值轨迹 + 各时效集合均值；箭头=均值−真值
# 协议：全中文标注；图内不加"XX图"标题；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})

# ===== 1. 合成示例数据：Hénon 吸引子 + 扰动集合均值 =====
def henon_traj(x0, y0, n, a=1.4, b=0.3):
    xs, ys = [x0], [y0]
    for _ in range(n):
        x0, y0 = 1 - a * x0 * x0 + y0, b * x0
        xs.append(x0)
        ys.append(y0)
    return np.array(xs), np.array(ys)

rng = np.random.default_rng(10)
N = 320
bx, by = henon_traj(0.1, 0.1, N)
all_leads = np.arange(1, 26)
shown_leads = [1, 6, 23]
origins = np.arange(40, 220, 9)
n_ens = 35
vectors = {}
mag_by_lead = []
for k in all_leads:
    vecs = []
    for i in origins:
        endpoints = []
        for _ in range(n_ens):
            px = bx[i] + rng.normal(0, 0.004)
            py = by[i] + rng.normal(0, 0.004)
            ex, ey = henon_traj(px, py, int(k))
            endpoints.append((ex[-1], ey[-1]))
        mean_xy = np.mean(endpoints, axis=0)
        true_xy = np.array([bx[i + k], by[i + k]])
        delta = mean_xy - true_xy
        vecs.append((*true_xy, *delta))
    arr = np.asarray(vecs)
    vectors[int(k)] = arr
    mag_by_lead.append(np.hypot(arr[:, 2], arr[:, 3]))

# ===== 2. 绘图 =====
fig = plt.figure(figsize=(13.2, 7.4))
gs = fig.add_gridspec(2, 3, height_ratios=[0.8, 1.25], hspace=0.34)
ax_top = fig.add_subplot(gs[0, :])
mag = np.asarray(mag_by_lead)
median = np.median(mag, axis=1)
lo, hi = np.quantile(mag, [0.10, 0.90], axis=1)
ax_top.plot(all_leads, median, color="#0072B2", lw=2, label="中位误差")
ax_top.fill_between(all_leads, lo, hi, color="#56B4E9", alpha=0.22,
                    linewidth=0, label="10%–90% 区间")
for k in shown_leads:
    ax_top.axvline(k, color="#999999", ls=":", lw=0.8)
ax_top.set_xlabel("预报时效")
ax_top.set_ylabel("二维误差幅度")
ax_top.legend(loc="upper left")

axes = [fig.add_subplot(gs[1, i]) for i in range(3)]
for ax, k in zip(axes, shown_leads):
    v = vectors[k]
    ax.scatter(bx, by, s=4, color="#C7D0D9", alpha=0.6, edgecolors="none")
    ax.quiver(v[:, 0], v[:, 1], v[:, 2], v[:, 3],
              angles="xy", scale_units="xy", scale=1,
              color="#D55E00", width=0.004, alpha=0.65)
    ax.set_title(f"第 {k} 步预报", fontsize=10.5)
    ax.set_xlabel("状态 x")
    ax.set_ylabel("状态 y")
axes[0].text(0.02, 0.03, "灰点=状态背景；红箭头=第 k 步真值→第 k 步集合均值",
             transform=axes[0].transAxes, fontsize=8.5, color="#555")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("误差向量演化.png")
plt.close(fig)
print("已输出：误差向量演化.png")
```
