## 图表说明

把逐日数值铺进"行=月、列=日"的网格，色深即强度；每年一行，共享同一色标即可横向比较年度水平。季节循环、工作日/周末起伏、异常日（寒潮、热浪、节假日）全部在一张图内，适合日粒度总量的第一张总览图。注意：列只有 28–31 天有效，其余格留白；色标用感知均匀的 cividis，事件框建议白框与暖色框各表一类。

```python
# -*- coding: utf-8 -*-
# 图表模板：月×日 日历热力图（多年度堆叠）
# 数据契约：行=月(1-12)、列=日(1-31)、值=强度；多年度各一行；可选事件日描框
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT, C_GREY = "#0072B2", "#D55E00", "#6B6B6B"

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(42)

def synth_year(year: int) -> pd.Series:
    """合成一年逐日强度（千次）：年周期 + 工作日效应 + 噪声。"""
    idx = pd.date_range(f"{year}-01-01", f"{year}-12-31", freq="D")
    base = 60 + 45 * np.sin((idx.dayofyear.to_numpy() - 105) / 365 * 2 * np.pi)
    weekly = np.where(idx.dayofweek.to_numpy() < 5, 1.15, 0.75)
    vals = base * weekly * rng.normal(1, 0.08, len(idx)) * (1.0 if year == 2024 else 1.08)
    return pd.Series(np.clip(vals, 5, None), index=idx)

years = [2024, 2025]
series = {y: synth_year(y) for y in years}
vmax = max(s.max() for s in series.values())
snow = pd.to_datetime(["2024-01-17", "2024-02-13", "2025-01-09", "2025-02-06"])
heat = pd.to_datetime(["2024-07-16", "2024-07-22", "2025-07-11", "2025-08-02"])

# ===== 2. 绘图 =====
fig, axes = plt.subplots(2, 1, figsize=(11, 5.2))
for ax, y in zip(axes, years):
    s = series[y]
    mat = np.full((12, 31), np.nan)
    mat[s.index.month - 1, s.index.day - 1] = s.to_numpy()
    im = ax.imshow(mat, aspect="auto", cmap="cividis", vmin=0, vmax=vmax)
    ax.set_yticks(range(12), [f"{m}月" for m in range(1, 13)], fontsize=8)
    ax.set_xticks([0, 6, 12, 18, 24, 30], ["1", "7", "13", "19", "25", "31"], fontsize=8)
    ax.set_ylabel(f"{y} 年")
    ax.grid(False)
    for d in snow[snow.year == y]:
        ax.add_patch(plt.Rectangle((d.day - 1.42, d.month - 1.42), 0.84, 0.84,
                                   fill=False, ec="white", lw=1.1))
    for d in heat[heat.year == y]:
        ax.add_patch(plt.Rectangle((d.day - 1.42, d.month - 1.42), 0.84, 0.84,
                                   fill=False, ec=C_ACCENT, lw=1.1))
cb = fig.colorbar(im, ax=axes, shrink=0.85, pad=0.012)
cb.set_label("每日强度（千次）")
axes[1].set_xlabel("每月日期")
axes[0].plot([], [], marker="s", ls="", ms=7, mfc="white", mec="#555", label="降雪日")
axes[0].plot([], [], color=C_ACCENT, lw=1.5, label="热浪日")
axes[0].legend(loc="lower left", bbox_to_anchor=(0, 1.02), ncols=2)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("日历热力图.png")
plt.close(fig)
print("已输出：日历热力图.png")
```
