## 图表说明

因果发现结果的网络化呈现：每条检出边一条弧，箭头指方向，弧宽编码证据强度（技能差），双色区分"与外部证据一致/不一致"。本模板用环形抽象布局，任何节点对都可以画；若节点有经纬度，把坐标字典换成真实点位即可叠加地图。注意：节点多于 15 个时弧线会糊，先按强度筛 top 边；弧的隆起方向统一，避免正反两条边完全重合。

```python
# -*- coding: utf-8 -*-
# 图表模板：方向性网络弧线图（贝塞尔弧 + 箭头，抽象环形布局）
# 数据契约：边表(源,目标,方向强度,是否可信)；节点坐标=抽象布局或真实经纬
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
from matplotlib.patches import FancyArrowPatch

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT = "#0072B2", "#D55E00"

# ===== 1. 合成示例数据（使用时替换为真实输出） =====
rng = np.random.default_rng(27)
names = [f"单元{chr(65 + i)}" for i in range(10)]
ang = np.linspace(0, 2 * np.pi, len(names), endpoint=False)
xy = {nm: (np.cos(a), np.sin(a)) for nm, a in zip(names, ang)}   # 环形布局；可换真实经纬

pairs = [("单元A", "单元B", .32, True), ("单元B", "单元C", .18, True),
         ("单元C", "单元D", .11, False), ("单元D", "单元E", .27, True),
         ("单元E", "单元F", .15, True), ("单元F", "单元G", .08, False),
         ("单元G", "单元H", .21, True), ("单元H", "单元I", .13, True),
         ("单元I", "单元J", .24, True), ("单元J", "单元A", .19, True),
         ("单元A", "单元E", .09, False), ("单元B", "单元G", .14, True)]
edges = pd.DataFrame(pairs, columns=["a", "b", "skill_gap", "match"])

# ===== 2. 绘图 =====
fig, ax = plt.subplots(figsize=(8.6, 7.4))
wmax = edges["skill_gap"].max()
for _, r in edges.iterrows():
    p0, p1 = np.array(xy[r["a"]]), np.array(xy[r["b"]])
    cc = C_MAIN if r["match"] else C_ACCENT
    lw = 0.7 + 3.2 * r["skill_gap"] / wmax
    reciprocal = ((edges["a"].eq(r["b"])) & (edges["b"].eq(r["a"]))).any()
    rad = (0.23 if r["a"] < r["b"] else -0.23) if reciprocal else 0.14
    arrow = FancyArrowPatch(
        p0, p1, connectionstyle=f"arc3,rad={rad}",
        arrowstyle="-|>", mutation_scale=10 + 1.5 * lw,
        linewidth=lw, color=cc, alpha=0.68,
        shrinkA=10, shrinkB=10, zorder=2,
    )
    ax.add_patch(arrow)
for nm, (x, y) in xy.items():
    ax.scatter([x], [y], s=52, color="#333", zorder=5)
    ax.text(x * 1.16, y * 1.16, nm, ha="center", va="center", fontsize=9.5)
ax.set_xlim(-1.35, 1.35)
ax.set_ylim(-1.35, 1.35)
ax.set_aspect("equal")
ax.set_axis_off()
ax.plot([], [], color=C_MAIN, lw=2.5, label="方向一致（线宽∝强度差）")
ax.plot([], [], color=C_ACCENT, lw=2.5, label="方向不一致（线宽∝强度差）")
ax.legend(loc="lower left", fontsize=9)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("方向网络弧线图.png")
plt.close(fig)
print("已输出：方向网络弧线图.png")
```
