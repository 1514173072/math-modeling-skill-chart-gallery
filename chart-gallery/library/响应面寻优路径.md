## 图表说明

底图回答“目标函数地形是什么”，连线回答“算法如何走到最终点”。路径只能来自真实迭代日志；若只有最终参数，不应伪造寻优轨迹。颜色条必须写明指标及越大或越小越优。

```python
# -*- coding: utf-8 -*-
# 数据契约：参数1×参数2网格目标值 + 真实迭代路径
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73","#D55E00","#CC79A7","#56B4E9","#F0E442","#999999"]
x=np.linspace(-3,4,120); y=np.linspace(-3,3,100); X,Y=np.meshgrid(x,y)
Z=1.8-(X-1.1)**2/8-(Y+0.4)**2/5+0.28*np.sin(1.7*X)*np.cos(1.3*Y)
path=np.array([[-2.5,2.3],[-1.6,1.7],[-.8,.9],[-.1,.35],[.45,-.05],[.82,-.28],[1.05,-.42]])
fig,ax=plt.subplots(figsize=(7.4,5.5))
levels=np.linspace(Z.min(),Z.max(),16)
cf=ax.contourf(X,Y,Z,levels=levels,cmap="viridis")
ax.contour(X,Y,Z,levels=levels[::2],colors="white",linewidths=.45,alpha=.65)
ax.plot(path[:,0],path[:,1],"-o",color=OI[1],lw=2,ms=4,label="迭代路径")
for i,(px,py) in enumerate(path[::2],1): ax.text(px+.08,py+.08,str(2*i-1),color="white",fontsize=8,weight="bold")
ax.scatter(path[-1,0],path[-1,1],s=120,marker="*",color=OI[3],edgecolor="white",lw=.9,zorder=5,label="最终方案")
ax.set_xlabel("参数 $x_1$"); ax.set_ylabel("参数 $x_2$"); ax.legend(loc="lower left")
cbar=fig.colorbar(cf,ax=ax,pad=.02); cbar.set_label("目标函数值（越大越好）")
fig.savefig("响应面寻优路径.png"); plt.close(fig)
print("已输出：响应面寻优路径.png")
```
