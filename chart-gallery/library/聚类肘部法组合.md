## 图表说明

聚类分析同时交代两类选 K 证据：SSE 只随 K 增大而下降，用于观察边际收益；轮廓系数用于给出可计算的候选值。本模板用轮廓系数最大值选择 K，并在第三面板显示 PCA 二维投影。投影只用于展示，不是聚类输入；真实研究仍应结合稳定性、可解释性和业务约束，不应机械采用单一指标。

```python
# -*- coding: utf-8 -*-
# 图表模板：聚类肘部法 + 结果散点（双面板）
# 数据契约：数值特征矩阵（标准化后聚类）；K 的 SSE 序列 + 两维散点按簇着色
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from sklearn.cluster import KMeans
from sklearn.datasets import make_blobs
from sklearn.decomposition import PCA
from sklearn.metrics import silhouette_score

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实标准化特征矩阵） =====
rng = np.random.default_rng(5)
X, _ = make_blobs(n_samples=600, n_features=6, centers=4, cluster_std=1.1, random_state=5)

# ===== 2. 选 K：SSE + 轮廓系数 =====
ks = list(range(2, 10))
fits = [KMeans(n_clusters=k, random_state=42, n_init=10).fit(X) for k in ks]
sse = [fit.inertia_ for fit in fits]
sil = [silhouette_score(X, fit.labels_) for fit in fits]
k_best = ks[int(np.argmax(sil))]
best_fit = fits[ks.index(k_best)]

# ===== 3. 绘图 =====
fig, axes = plt.subplots(1, 3, figsize=(13.2, 4.3))
ax = axes[0]
ax.plot(ks, sse, "-o", color=OI[0], lw=2)
ax.axvline(k_best, color=OI[3], ls="--", lw=1.2)
ax.set_xlabel("簇数 K")
ax.set_ylabel("簇内平方和 SSE")
ax.set_xticks(ks)

ax = axes[1]
ax.plot(ks, sil, "-o", color=OI[2], lw=2)
ax.axvline(k_best, color=OI[3], ls="--", lw=1.2)
ax.scatter([k_best], [max(sil)], s=55, color=OI[3], zorder=3)
ax.text(k_best, max(sil), f"  选 K={k_best}", va="bottom", fontsize=9)
ax.set_xlabel("簇数 K")
ax.set_ylabel("轮廓系数")
ax.set_xticks(ks)

ax = axes[2]
labels = best_fit.labels_
proj = PCA(n_components=2).fit_transform(X)
for i in range(k_best):
    m = labels == i
    ax.scatter(proj[m, 0], proj[m, 1], s=14, alpha=0.6, color=OI[i % len(OI)],
               edgecolors="none", label=f"簇 {i + 1}")
ax.set_xlabel("PCA 第一维（仅展示）")
ax.set_ylabel("PCA 第二维（仅展示）")
ax.legend(markerscale=1.6)

# ===== 4. 输出（文件名必须与笔记同名） =====
fig.savefig("聚类肘部法组合.png")
plt.close(fig)
print("已输出：聚类肘部法组合.png")
```
