## 图表说明

左图展示局部热点统计量，右图展示同一位置的标准误，并用斜线标记“不确定性过高、暂不解释”的区域。示例使用规则网格以保持自包含；替换为行政区或其他空间单元时，必须保留投影、邻接权重、显著性校正和边界来源信息。

```python
# -*- coding: utf-8 -*-
# 数据契约：空间单元坐标/边界 + 热点统计量 + 标准误或置信度
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"savefig.dpi":300,"savefig.bbox":"tight"})
rng=np.random.default_rng(5); x=np.linspace(0,10,55); y=np.linspace(0,7,42); X,Y=np.meshgrid(x,y)
hot=2.8*np.exp(-((X-7.3)**2+(Y-2.1)**2)/3)-2.2*np.exp(-((X-2.3)**2+(Y-5.2)**2)/2.4)+.22*rng.normal(size=X.shape)
se=.18+.62*np.exp(-((X-5.3)**2+(Y-4.2)**2)/4)+.12*rng.random(X.shape)
fig,(a,b)=plt.subplots(1,2,figsize=(10.6,4.5),sharex=True,sharey=True)
im=a.contourf(X,Y,hot,levels=np.linspace(-2.5,2.8,15),cmap="RdBu_r",extend="both")
a.contour(X,Y,hot,levels=[-1.96,1.96],colors="black",linewidths=.9)
cb=fig.colorbar(im,ax=a,pad=.02); cb.set_label("局部热点统计量 z")
im2=b.contourf(X,Y,se,levels=12,cmap="YlOrBr")
uncertain=se>.58; b.contourf(X,Y,uncertain,levels=[.5,1.5],colors="none",hatches=["////"])
cb2=fig.colorbar(im2,ax=b,pad=.02); cb2.set_label("估计标准误")
for ax,label in [(a,"效应强度"),(b,"不确定性（斜线区慎解释）")]:
    ax.set_xlabel("空间横坐标"); ax.set_ylabel("空间纵坐标"); ax.text(.02,.98,label,transform=ax.transAxes,ha="left",va="top",bbox={"facecolor":"white","alpha":.82,"edgecolor":"none"})
fig.savefig("空间热点与不确定性.png"); plt.close(fig)
print("已输出：空间热点与不确定性.png")
```
