## 图表说明

蓝色区域是同时满足全部硬约束的可行域，虚线为目标函数等值线，橙色星号为最大化目标的网格解。图适合解释二维模型；高维问题只能画切片，并应写清其他变量固定值，不能把二维投影误称为完整可行域。

```python
# -*- coding: utf-8 -*-
# 数据契约：两个决策变量 + 目标函数 + 全部硬约束
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#D9D9D9","savefig.dpi":300,"savefig.bbox":"tight"})
X,Y=np.meshgrid(np.linspace(0,10,401),np.linspace(0,9,361))
feasible=(2*X+Y<=14)&(X+2*Y<=13)&(X>=1)&(Y>=1)&(Y<=.9*X+2)
objective=4.2*X+3.4*Y-.18*(X**2+Y**2)
masked=np.where(feasible,objective,-np.inf); k=np.unravel_index(np.argmax(masked),masked.shape); ox,oy=X[k],Y[k]
fig,ax=plt.subplots(figsize=(7.2,5.5))
ax.contourf(X,Y,feasible.astype(int),levels=[.5,1.5],colors=["#56B4E9"],alpha=.28)
ax.contour(X,Y,2*X+Y,levels=[14],colors=["#555555"],linewidths=1.4)
ax.contour(X,Y,X+2*Y,levels=[13],colors=["#0072B2"],linewidths=1.4)
cs=ax.contour(X,Y,objective,levels=7,colors="#8A8A8A",linestyles="--",linewidths=.8); ax.clabel(cs,fontsize=8,fmt="%.0f")
ax.scatter([ox],[oy],marker="*",s=170,color="#E69F00",edgecolor="white",zorder=5)
ax.annotate(f"最优解 ({ox:.2f}, {oy:.2f})\n目标值 {objective[k]:.2f}",(ox,oy),xytext=(18,-35),textcoords="offset points",arrowprops={"arrowstyle":"->","color":"#A45D00"})
ax.set(xlim=(0,10),ylim=(0,9),xlabel="决策变量 $x_1$",ylabel="决策变量 $x_2$")
fig.savefig("可行域与最优解.png"); plt.close(fig)
print("已输出：可行域与最优解.png")
```
