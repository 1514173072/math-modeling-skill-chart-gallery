## 图表说明

ROC 衡量排序能力，PR 更适合正类稀少的数据；阈值面板用于确定实际决策点，混淆矩阵展示该阈值带来的错误结构。只能输入严格样本外预测概率，训练集曲线不能代表泛化性能。

```python
# -*- coding: utf-8 -*-
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from sklearn.metrics import roc_curve,roc_auc_score,precision_recall_curve,average_precision_score,confusion_matrix

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.2,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#DDDDDD","grid.linewidth":.6,"axes.axisbelow":True,"savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73"]
rng=np.random.default_rng(31); n=900; y=(rng.random(n)<.18).astype(int)
scores={
    "逻辑回归":np.clip(.15+.48*y+rng.normal(0,.22,n),0,1),
    "随机森林":np.clip(.12+.58*y+rng.normal(0,.19,n),0,1),
    "提升树":np.clip(.10+.66*y+rng.normal(0,.17,n),0,1),
}
fig,axs=plt.subplots(2,2,figsize=(10.6,7.7)); axr,axp,axt,axc=axs.ravel()
for (name,s),c in zip(scores.items(),OI):
    fpr,tpr,_=roc_curve(y,s); pre,rec,_=precision_recall_curve(y,s)
    axr.plot(fpr,tpr,color=c,lw=1.8,label=f"{name}  AUC={roc_auc_score(y,s):.3f}")
    axp.plot(rec,pre,color=c,lw=1.8,label=f"{name}  AP={average_precision_score(y,s):.3f}")
axr.plot([0,1],[0,1],ls="--",color="#999999",lw=1); axr.set(xlabel="假阳性率",ylabel="真阳性率",title="ROC：整体排序能力"); axr.legend(fontsize=8.5)
axp.axhline(y.mean(),ls="--",color="#999999",lw=1,label=f"正类基线={y.mean():.1%}"); axp.set(xlabel="召回率",ylabel="精确率",title="PR：不平衡数据重点指标"); axp.legend(fontsize=8.5)

best=scores["提升树"]; pre,rec,thr=precision_recall_curve(y,best); pre,rec=pre[:-1],rec[:-1]; f1=2*pre*rec/(pre+rec+1e-12); k=np.argmax(f1); cut=thr[k]
axt.plot(thr,pre,label="精确率",color=OI[0]); axt.plot(thr,rec,label="召回率",color=OI[1]); axt.plot(thr,f1,label="F1",color=OI[2],lw=2)
axt.axvline(cut,color="#D55E00",ls="--",label=f"F1最优阈值={cut:.2f}"); axt.set(xlabel="决策阈值",ylabel="指标值",title="阈值—业务指标权衡"); axt.legend(fontsize=8.5)
cm=confusion_matrix(y,best>=cut); im=axc.imshow(cm,cmap=mpl.colors.LinearSegmentedColormap.from_list("cm",["#F3F7FA","#0072B2"]))
for i in range(2):
    for j in range(2): axc.text(j,i,str(cm[i,j]),ha="center",va="center",fontsize=14,color="white" if cm[i,j]>cm.max()*.55 else "#222222")
axc.set_xticks([0,1],["预测负类","预测正类"]); axc.set_yticks([0,1],["实际负类","实际正类"]); axc.set_title(f"混淆矩阵（阈值={cut:.2f}）"); axc.grid(False)
fig.subplots_adjust(wspace=.28,hspace=.32); fig.savefig("分类阈值诊断面板.png"); plt.close(fig)
print("已输出：分类阈值诊断面板.png")
```
