## 图表说明

类别名长、又需要同时比较总规模和内部构成时使用：左面板只比较总量，右面板统一换算为 100% 构成，因此不会把"条更长"误读成"某一组成占比更高"。Top-N 依据总量筛选并保持两面板行序一致。分组超过 6 个时应先合并长尾为"其他"。

```python
# -*- coding: utf-8 -*-
# 图表模板：总量与构成双面板横条图（Top-N）
# 数据契约：长表三列(类别,分组,数量)；透视后按合计取 Top-N
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(42)
industries = [f"类别{i:02d}" for i in range(1, 21)]
groups = ["分组甲", "分组乙", "分组丙", "分组丁"]
rows = [(i, g, int(rng.integers(20, 400))) for i in industries for g in groups]
df = pd.DataFrame(rows, columns=["类别", "分组", "数量"])

# ===== 2. 透视 + Top-N =====
pivot = df.pivot_table(index="类别", columns="分组", values="数量",
                       aggfunc="sum", fill_value=0)
top = pivot.sum(axis=1).sort_values(ascending=False).head(15).index
pivot = pivot.loc[top]

# ===== 3. 绘图 =====
totals = pivot.sum(axis=1)
shares = pivot.div(totals, axis=0) * 100
y = np.arange(len(pivot))
fig, (ax0, ax1) = plt.subplots(
    1, 2, figsize=(11.4, 7), sharey=True,
    gridspec_kw={"width_ratios": [0.72, 1.55], "wspace": 0.06},
)
ax0.barh(y, totals, color="#7A8A99", height=0.72)
ax0.set_yticks(y, pivot.index)
ax0.invert_yaxis()
ax0.set_xlabel("总量")
ax0.set_ylabel("类别")
for yi, total in zip(y, totals):
    ax0.text(total, yi, f" {int(total)}", va="center", fontsize=8.2, color="#555555")

left = np.zeros(len(pivot))
for col, cc in zip(shares.columns, OI):
    ax1.barh(y, shares[col], left=left, color=cc, label=col, height=0.72)
    left += shares[col].to_numpy()
ax1.set_xlim(0, 100)
ax1.set_xlabel("组内构成（%）")
ax1.legend(ncols=2, loc="lower right")

# ===== 4. 输出（文件名必须与笔记同名） =====
fig.savefig("堆叠横条图.png")
plt.close(fig)
print("已输出：堆叠横条图.png")
```
