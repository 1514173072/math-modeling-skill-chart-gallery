## 图表说明

参数扫描遇到"两个量纲都要看、且在同一位置发生突变"时用这张：主序参量与辅助指标分别放在上下两个面板，共享控制参数轴，垂直参考线帮助判断两者是否在同一区域发生变化。相比双 y 轴，分面不会因为人为设置轴范围而夸大同步性。单一指标陡降只能作为候选转折证据，是否属于机制切换仍需结合理论、重复试验或其他诊断。与《超参稳健性曲线》分工：那张关注"选哪点稳"，这张关注"两个指标在哪个参数区间同时变化"。

```python
# -*- coding: utf-8 -*-
# 图表模板：共享横轴参数扫描图（双指标分面）
# 数据契约：控制参数网格 × 两条量纲不同的响应序列
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG

import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
OI = ["#0072B2", "#E69F00", "#009E73", "#D55E00", "#CC79A7", "#56B4E9", "#F0E442", "#999999"]
C_L, C_R = OI[3], OI[0]

# ===== 1. 合成示例数据（使用时替换为真实扫描结果） =====
r = np.linspace(3.5, 4.0, 600)
sig = lambda z: 1 / (1 + np.exp(-z))
order = 0.72 - 0.26 * sig((r - 3.67) / 0.004) - 0.39 * sig((r - 3.845) / 0.005)   # 序参量
order += np.random.default_rng(3).normal(0, 0.004, len(r))
cpl = 4.6 - 1.8 * sig((r - 3.67) / 0.02) + 1.1 * sig((r - 3.845) / 0.03)          # 第二指标

# ===== 2. 绘图（共享横轴双分面） =====
fig, (ax, ax2) = plt.subplots(
    2, 1, figsize=(9.2, 6.2), sharex=True,
    gridspec_kw={"height_ratios": [1, 1], "hspace": 0.10},
)
ax.plot(r, order, color=C_L, lw=2)
ax.set_ylabel("序参量 μ")
ax.set_ylim(0, 0.85)

ax2.plot(r, cpl, color=C_R, lw=2)
ax2.set_ylabel("平均路径长度")
ax2.set_xlabel("控制参数 r")
ax2.set_ylim(2.4, 5.0)

for panel in (ax, ax2):
    panel.axvline(3.67, color="#666666", ls="--", lw=0.9)
    panel.axvline(3.85, color="#999999", ls=":", lw=0.9)

ax.text(3.674, 0.80, "候选临界点", color="#555555", fontsize=9, va="top")
ax2.text(3.854, 2.53, "候选窗口", color="#666666", fontsize=9, va="bottom")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("双轴参数扫描.png")
plt.close(fig)
print("已输出：双轴参数扫描.png")
```
