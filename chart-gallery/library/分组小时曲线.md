## 图表说明

日型分别成面板，颜色只编码群体，阴影表示预先定义的不确定性区间。这样既能比较同一日型下的群体差异，也能沿共享坐标比较跨日型节奏，并避免六条线叠在一起。峰值位置由数据计算；阴影究竟是标准误、置信区间还是分位数区间，必须在正文明确。

```python
# -*- coding: utf-8 -*-
# 图表模板：分组小时曲线（日型分面与不确定性带）
# 数据契约：日型×群体×小时的均值与区间上下限
# 协议：全中文标注；图内不加"XX图"标题；Okabe-Ito 色板；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})
C_MAIN, C_ACCENT = "#0072B2", "#D55E00"

# ===== 1. 合成示例数据（使用时替换为真实数据） =====
rng = np.random.default_rng(4)
hours = np.arange(24)
rows = []
for dt, wk in [("工作日", 1.0), ("周六", 0.85), ("周日", 0.75)]:
    for grp, base in [("会员", 1.15), ("散客", 0.8)]:
        if dt == "工作日":
            shape = 9 * np.exp(-((hours - 8) ** 2) / 4.5) + 11 * np.exp(-((hours - 17.5) ** 2) / 6) + 2
        else:
            shape = 7.5 * np.exp(-((hours - 13.5) ** 2) / 20) + 2.2
        mean = shape * base * wk * rng.normal(1, 0.03, 24)
        half = 0.08 * mean + 0.18
        for h, v, e in zip(hours, mean, half):
            rows.append((dt, grp, h, v, v - e, v + e))
g = pd.DataFrame(rows, columns=["dow_type", "group", "hour", "mean", "lo", "hi"])

# ===== 2. 绘图 =====
day_types = ["工作日", "周六", "周日"]
colors = {"会员": C_MAIN, "散客": C_ACCENT}
fig, axes = plt.subplots(1, 3, figsize=(12, 4.1), sharex=True, sharey=True)
for ax, dt in zip(axes, day_types):
    for grp in ["会员", "散客"]:
        sub = g[(g["dow_type"] == dt) & (g["group"] == grp)]
        x = sub["hour"].to_numpy()
        mean = sub["mean"].to_numpy()
        ax.plot(x, mean, color=colors[grp], lw=2, label=grp)
        ax.fill_between(x, sub["lo"].to_numpy(), sub["hi"].to_numpy(),
                        color=colors[grp], alpha=0.14, linewidth=0)
    peak_row = g[g["dow_type"].eq(dt)].loc[g[g["dow_type"].eq(dt)]["mean"].idxmax()]
    ax.scatter(peak_row["hour"], peak_row["mean"], s=24, color="#333333", zorder=3)
    ax.text(peak_row["hour"], peak_row["mean"], f"  峰值 {int(peak_row['hour'])} 时",
            va="bottom", fontsize=8.3, color="#444444")
    ax.set_title(dt, fontsize=10.5)
    ax.set_xticks(range(0, 25, 4))
    ax.set_xlabel("小时")
axes[0].set_ylabel("均值强度（阴影为 95% 区间）")
axes[-1].legend(loc="upper right")

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("分组小时曲线.png")
plt.close(fig)
print("已输出：分组小时曲线.png")
```
