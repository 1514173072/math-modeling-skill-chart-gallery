## 图表说明

树状图回答“样本如何逐级合并”，簇中心热力图回答“分成这些簇后各簇有什么特征”。切树数量应来自业务需求、轮廓系数或稳定性分析，而不是只看图随意决定；右图只能使用与左图同一标准化口径的数据。

```python
# -*- coding: utf-8 -*-
# 数据契约：样本×连续指标矩阵；固定标准化、距离、连接法与切树规则
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
from scipy.cluster.hierarchy import dendrogram, fcluster, linkage
from sklearn.preprocessing import StandardScaler

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
rng=np.random.default_rng(9); vars=["需求强度","响应速度","资源占用","服务质量","风险"]
means=np.array([[1.2,.8,-.5,.9,-.7],[-.6,.1,1.1,-.2,.8],[.1,-1.0,.2,.4,.1]])
X=np.vstack([rng.normal(m,.45,(12,len(vars))) for m in means]); Z=StandardScaler().fit_transform(X)
L=linkage(Z,method="ward"); labels=fcluster(L,t=3,criterion="maxclust")
centers=np.vstack([Z[labels==k].mean(0) for k in range(1,4)])
fig,(ax,ax2)=plt.subplots(1,2,figsize=(11,5.3),gridspec_kw={"width_ratios":[1.25,1]})
dendrogram(L,ax=ax,no_labels=True,color_threshold=L[-2,2],above_threshold_color="#808080")
ax.set(xlabel="样本（按聚类顺序）",ylabel="Ward 合并距离")
im=ax2.imshow(centers,cmap="RdBu_r",vmin=-1.5,vmax=1.5,aspect="auto")
ax2.set_xticks(range(len(vars)),vars,rotation=35,ha="right"); ax2.set_yticks(range(3),[f"簇{k}（n={(labels==k).sum()}）" for k in range(1,4)])
for i in range(3):
    for j in range(len(vars)): ax2.text(j,i,f"{centers[i,j]:.2f}",ha="center",va="center",fontsize=8,color="white" if abs(centers[i,j])>.75 else "black")
cbar=fig.colorbar(im,ax=ax2,pad=.03); cbar.set_label("簇中心标准分")
fig.savefig("聚类树与簇画像.png"); plt.close(fig)
print("已输出：聚类树与簇画像.png")
```
