## 图表说明

在固定误差阈值处，曲线越高说明达标样本比例越大；同一累计概率下，曲线越靠左越好。必须保证各模型对应同一测试集，并同时报告样本量和关键分位数，避免仅凭局部交叉位置下结论。

```python
# -*- coding: utf-8 -*-
# 数据契约：模型标签 + 同口径测试样本的非负误差
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.5,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#D9D9D9","savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73","#D55E00","#CC79A7","#56B4E9","#F0E442","#999999"]
rng=np.random.default_rng(21)
errors={"模型甲":rng.lognormal(-.55,.50,260),"模型乙":rng.lognormal(-.38,.58,260),"模型丙":rng.lognormal(-.25,.42,260)}
threshold=.85
fig,ax=plt.subplots(figsize=(7.4,5.1))
for (name,e),color in zip(errors.items(),OI):
    x=np.sort(e); y=np.arange(1,len(x)+1)/len(x)
    rate=(e<=threshold).mean(); q90=np.quantile(e,.9)
    ax.step(x,y,where="post",lw=2,color=color,label=f"{name}（n={len(e)}, P90={q90:.2f}）")
    ax.scatter([threshold],[rate],s=28,color=color,zorder=4)
ax.axvline(threshold,color="#555555",ls="--",lw=1.1)
ax.text(threshold+.03,.05,f"业务阈值 {threshold}",rotation=90,va="bottom",color="#555555")
ax.set(xlabel="绝对误差",ylabel="累计样本比例",ylim=(0,1.01)); ax.legend(loc="lower right")
fig.savefig("误差经验分布比较.png"); plt.close(fig)
print("已输出：误差经验分布比较.png")
```
