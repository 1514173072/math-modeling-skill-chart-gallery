## 图表说明

"结果对人为阈值有多敏感"的完整交代：左边是选定阈值下的状态构成（堆叠横条，段内直接标数），右边把激活/持续两个阈值扫成网格，两条主状态的数量曲线平铺在图上——曲线平缓说明结论不挑阈值，陡变处就是结论的脆弱点；第二个阈值用横向微错位（0.06×阈值）编码，一轴装两参数。适合任何需要预设截止点的分窗/分状态分析。注意：错位系数要写明，否则读者以为横轴失真。

```python
# -*- coding: utf-8 -*-
# 图表模板：状态构成堆叠横条 + 阈值敏感性曲线
# 数据契约：状态×分组计数表 + 两参数阈值网格×各状态数量
# 协议：全中文标注；图内不加"XX图"标题；300dpi；末尾输出与笔记同名 PNG
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

mpl.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Source Han Sans SC", "Noto Sans CJK SC", "Microsoft YaHei", "SimHei"],
    "axes.unicode_minus": False,
    "font.size": 10.5, "axes.labelsize": 10.5,
    "xtick.labelsize": 9.5, "ytick.labelsize": 9.5, "legend.fontsize": 9.5,
    "axes.spines.top": False, "axes.spines.right": False,
    "axes.grid": True, "grid.color": "#D9D9D9", "grid.linewidth": 0.6,
    "axes.axisbelow": True, "figure.dpi": 110,
    "savefig.dpi": 300, "savefig.bbox": "tight", "legend.frameon": False,
})

# ===== 1. 合成示例数据（使用时替换为真实计数） =====
rng = np.random.default_rng(5)
states = ["未激活", "短期激活", "持续复苏", "激活后衰退", "复苏后再次衰退", "数据不足"]
colors = ["#7b8794", "#d9953f", "#348c83", "#b95858", "#6e5aa6", "#b8bec5"]
counts = pd.DataFrame({
    "状态": states * 2,
    "分组": ["英文"] * 6 + ["简体中文"] * 6,
    "数量": [420, 110, 180, 90, 60, 40, 310, 150, 120, 140, 80, 50],
})
acts = np.round(np.arange(0.05, 0.32, 0.05), 2)          # 阈值一：激活
suss = np.round(np.arange(0.30, 0.82, 0.10), 2)          # 阈值二：持续
base_a, base_s = 260, 180
sens = []
for lang, mult in [("英文", 1.0), ("简体中文", 0.82)]:
    for a in acts:
        for s in suss:
            short = base_a * mult * np.exp(-5 * (a - 0.05)) * rng.normal(1, 0.03)
            sus = base_s * mult * (1 - np.exp(-6 * (s - 0.25))) * rng.normal(1, 0.03)
            sens.append((lang, a, s, max(short, 5), max(sus, 5)))
sens = pd.DataFrame(sens, columns=["分组", "激活阈值", "持续阈值", "短期激活", "持续复苏"])

# ===== 2. 绘图 =====
fig, axes = plt.subplots(1, 2, figsize=(15, 7), gridspec_kw={"width_ratios": [1.05, 1.4]},
                         constrained_layout=True)
ax = axes[0]
pivot = counts.pivot(index="状态", columns="分组", values="数量").reindex(states).fillna(0)
y = np.arange(len(states))
left = np.zeros(len(states))
for lang, color in [("英文", "#2f6f9f"), ("简体中文", "#c46b37")]:
    values = pivot.get(lang, pd.Series(0, index=pivot.index)).to_numpy()
    ax.barh(y, values, left=left, color=color, label=lang)
    for pos, v, b in zip(y, values, left):
        if v >= 3:
            ax.text(b + v / 2, pos, f"{int(v)}", ha="center", va="center",
                    color="white", fontsize=9)
    left += values
ax.set_yticks(y, states)
ax.set_xlabel("单元数")
ax.set_ylabel("状态")
ax.invert_yaxis()
ax.legend(frameon=False)
ax.grid(axis="x", color="#e8e8e8", lw=0.6)
ax.spines[["top", "right"]].set_visible(False)

ax = axes[1]
for lang, color, marker in [("英文", "#2f6f9f", "o"), ("简体中文", "#c46b37", "s")]:
    sub = sens.loc[sens["分组"].eq(lang)]
    for state, ls in [("持续复苏", "-"), ("短期激活", "--")]:
        grid = (sub.groupby(["激活阈值", "持续阈值"], as_index=False)[state].sum())
        grid["x"] = grid["激活阈值"] + 0.06 * grid["持续阈值"]   # 第二参数错位编码
        grid = grid.sort_values(["激活阈值", "持续阈值"])
        ax.plot(grid["x"], grid[state], marker=marker, ms=3.5, lw=1.3,
                color=color, ls=ls, label=f"{lang}·{state}")
ax.set_ylim(0, None)
ax.set_xlabel("激活阈值（横轴；持续阈值作轻微横向错位）")
ax.set_ylabel("状态单元数")
ax.grid(color="#e8e8e8", lw=0.6)
ax.spines[["top", "right"]].set_visible(False)
ax.legend(frameon=False, fontsize=8, ncol=2)

# ===== 3. 输出（文件名必须与笔记同名） =====
fig.savefig("状态构成与阈值敏感性.png")
plt.close(fig)
print("已输出：状态构成与阈值敏感性.png")
```
