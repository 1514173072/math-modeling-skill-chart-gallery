## 图表说明

左侧显示响应是否线性、单调或存在阈值，右侧用龙卷风图排序 ±20% 扰动的影响范围。该图属于单因素局部敏感性分析，只能说明基准点附近的响应；不能替代 Sobol、Morris 等全局敏感性方法。

```python
# -*- coding: utf-8 -*-
import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np

mpl.rcParams.update({"font.family":"sans-serif","font.sans-serif":["Microsoft YaHei","SimHei"],"axes.unicode_minus":False,"font.size":10.3,"axes.spines.top":False,"axes.spines.right":False,"axes.grid":True,"grid.color":"#DDDDDD","grid.linewidth":.6,"axes.axisbelow":True,"savefig.dpi":300,"savefig.bbox":"tight","legend.frameon":False})
OI=["#0072B2","#E69F00","#009E73","#D55E00","#CC79A7","#56B4E9"]
names=["服务半径","站点容量","建设成本","需求强度","交通时间","人员配置"]
base=np.array([1.,1.,1.,1.,1.,1.]); delta=np.linspace(-.4,.4,17)
def model(p):
    return 82+6*np.log1p(p[0])+8*np.sqrt(p[1])-7*p[2]**1.35+9*np.tanh(1.5*(p[3]-.65))-5*p[4]**1.2+4*np.sin(1.2*p[5])+.9*p[0]*p[3]
y0=model(base); curves=[]
for j in range(len(names)):
    ys=[]
    for d in delta:
        p=base.copy(); p[j]*=1+d; ys.append(model(p))
    curves.append(np.array(ys))

fig,(ac,at)=plt.subplots(1,2,figsize=(11.0,5.8),gridspec_kw={"width_ratios":[1.15,.9]})
for name,ys,c in zip(names,curves,OI): ac.plot(delta*100,(ys/y0-1)*100,lw=1.8,color=c,label=name)
ac.axhline(0,color="#555555",lw=.8); ac.axvline(0,color="#555555",lw=.8); ac.set(xlabel="单参数相对扰动 / %",ylabel="模型输出相对变化 / %",title="连续扰动响应：识别非线性与阈值"); ac.legend(ncol=2,fontsize=8.5)

lo=[]; hi=[]
for j in range(len(names)):
    vals=[]
    for d in (-.2,.2):
        p=base.copy(); p[j]*=1+d; vals.append((model(p)/y0-1)*100)
    lo.append(min(vals)); hi.append(max(vals))
span=np.array(hi)-np.array(lo); order=np.argsort(span); y=np.arange(len(names));
at.barh(y,-np.array(lo)[order],left=np.array(lo)[order],color="#0072B2",alpha=.85,label="-20%情景"); at.barh(y,np.array(hi)[order],left=0,color="#D55E00",alpha=.82,label="+20%情景")
at.axvline(0,color="#333333",lw=.8); at.set_yticks(y,np.array(names)[order]); at.set(xlabel="输出相对基准变化 / %",title="龙卷风排序：±20%局部影响"); at.legend(loc="lower right",fontsize=8.5)
for yy,l,h in zip(y,np.array(lo)[order],np.array(hi)[order]): at.text(l-.05,yy,f"{l:.1f}%",ha="right",va="center",fontsize=8); at.text(h+.05,yy,f"{h:.1f}%",ha="left",va="center",fontsize=8)
at.set_xlim(min(lo)-.65,max(hi)+.55)
fig.subplots_adjust(wspace=.35); fig.savefig("扰动响应敏感度排序.png"); plt.close(fig)
print("已输出：扰动响应敏感度排序.png")
```
